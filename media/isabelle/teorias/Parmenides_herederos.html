Theory Parmenides_herederos

(* Parménides y sus herederos (23-10-2026). Sexto mundo de la mesa «Spinoza», a propuesta del profesor.
   Fuentes: Parménides, fragmentos B2, B3, B6, B7, B8 (Diels-Kranz); Meliso, B1-B7; Aristóteles, Refutaciones
   sofísticas 5, 167b13-20, y Física I 3, 186a10-16 (la falacia de Meliso); Zenón según Aristóteles, Física VI 9,
   239b11-14 (la dicotomía). Traducciones propias. Debate abierto: la lógica muestra qué premisas hacen qué. *)
theory Parmenides_herederos
  imports Main "HOL.Real" "HOL.Series"
begin

section ‹Parménides: el ser y el «es» (B2, B3, B6, B8)›

typedecl cosa
typedecl instante
consts Existe :: "cosa ⇒ instante ⇒ bool"   (* x es, en el sentido de «hay x», en t *)

(* B6.1-2: lo que es, es, y el no ser no es. Leído sobre instantes: nada de lo que es deja de ser ni empieza a ser. *)
definition P_no_ser :: bool where "P_no_ser ≡ ∀x t. Existe x t"

(* El salto que la mesa examina: leer «no ser F» como «no ser» sin más. Es lógica de orden superior: vale para
   cualquier propiedad P. Con P = «ser blanco en t», dejar de ser blanco sería dejar de ser; con P = «ser y»,
   ser distinto de y sería no ser. *)
definition P_lectura_existencial :: bool where
  "P_lectura_existencial ≡ ∀(P :: cosa ⇒ bool) x t. ¬ P x ⟶ ¬ Existe x t"

(* Los dos casos del salto que usa la mesa, en primer orden: «ser otro que y» leído como «no ser», y «no ser
   caliente» leído como «no ser». Los dos se siguen de la lectura existencial. *)
consts Caliente :: "cosa ⇒ instante ⇒ bool"
definition P_salto_otro :: bool where "P_salto_otro ≡ ∀x y t. x ≠ y ⟶ ¬ Existe x t"
definition P_salto_cambio :: bool where "P_salto_cambio ≡ ∀x t. ¬ Caliente x t ⟶ ¬ Existe x t"

lemma salto_casos:
  assumes "P_lectura_existencial"
  shows "P_salto_otro ∧ P_salto_cambio"
proof
  show "P_salto_otro" unfolding P_salto_otro_def
  proof (intro allI impI)
    fix x y :: cosa and t :: instante
    assume "x ≠ y"
    then have "¬ (λz. z = y) x" by simp
    then show "¬ Existe x t" using assms unfolding P_lectura_existencial_def by blast
  qed
  show "P_salto_cambio" unfolding P_salto_cambio_def
  proof (intro allI impI)
    fix x :: cosa and t :: instante
    assume "¬ Caliente x t"
    then have "¬ (λz. Caliente z t) x" by simp
    then show "¬ Existe x t" using assms unfolding P_lectura_existencial_def by blast
  qed
qed

(* En la mesa: con el no ser y el salto, el ser es uno y nada cambia. *)
theorem uno_mesa:
  assumes "P_no_ser" "P_salto_otro"
  shows "∀x y :: cosa. x = y"
  using assms unfolding P_no_ser_def P_salto_otro_def by blast

theorem inmovil_mesa:
  assumes "P_no_ser" "P_salto_cambio"
  shows "∀x t u. Caliente x t ⟷ Caliente x u"
  using assms unfolding P_no_ser_def P_salto_cambio_def by blast

(* B8.22-25: el ser es uno. Del no ser y del salto se sigue que no hay dos cosas. *)
theorem uno:
  assumes "P_no_ser" "P_lectura_existencial"
  shows "∀x y :: cosa. x = y"
proof (intro allI)
  fix x y :: cosa
  show "x = y"
  proof (rule ccontr)
    assume d: "x ≠ y"
    obtain t :: instante where True by simp
    have "¬ (λz. z = y) x" using d by simp
    then have "¬ Existe x t" using assms(2) unfolding P_lectura_existencial_def by blast
    moreover have "Existe x t" using assms(1) unfolding P_no_ser_def by blast
    ultimately show False by simp
  qed
qed

(* B8.26-41: el ser no cambia. Con F x t, «x es F en t», nada puede ser F en un instante y no serlo en otro. *)
theorem inmovil:
  fixes F :: "cosa ⇒ instante ⇒ bool"
  assumes "P_no_ser" "P_lectura_existencial"
  shows "∀x t. F x t"
proof (intro allI)
  fix x t
  show "F x t"
  proof (rule ccontr)
    assume n: "¬ F x t"
    have "¬ (λz. F z t) x" using n by simp
    then have "¬ Existe x t" using assms(2) unfolding P_lectura_existencial_def by blast
    moreover have "Existe x t" using assms(1) unfolding P_no_ser_def by blast
    ultimately show False by simp
  qed
qed

(* Sin el salto, el no ser no impide ni la pluralidad ni el cambio: dos cosas (los booleanos), que existen siempre,
   y una de ellas es F en un instante y no en otro. Es la salida de Platón en el Sofista (241d, el «parricidio»):
   «no ser» puede querer decir «ser otro», no «no ser en absoluto». *)
lemma sin_el_salto:
  fixes E :: "bool ⇒ nat ⇒ bool" and F :: "bool ⇒ nat ⇒ bool"
  defines "E ≡ λx t. True" and "F ≡ λx t. x ∧ t = 0"
  shows "(∀x t. E x t) ∧ True ≠ False ∧ F True 0 ∧ ¬ F True 1"
  unfolding E_def F_def by simp

section ‹Meliso: lo no generado no tiene principio (B1-B2)›

(* Aristóteles, Refutaciones sofísticas 5, 167b13-20: Meliso razona «lo generado tiene principio; el ser no es
   generado; luego no tiene principio». Es negar el antecedente: la conversión de la condicional no vale. *)
consts Generado :: "cosa ⇒ bool"
consts Principio :: "cosa ⇒ bool"
definition M_generado_principio :: bool where "M_generado_principio ≡ ∀x. Generado x ⟶ Principio x"

lemma falacia_de_Meliso:
  fixes G P :: "bool ⇒ bool"
  defines "G ≡ λx. False" and "P ≡ λx. True"
  shows "(∀x. G x ⟶ P x) ∧ ¬ G True ∧ P True"
  unfolding G_def P_def by simp

(* Con la conversa como premisa, el razonamiento sí vale: el fallo está en la premisa que falta. *)
theorem Meliso_con_la_conversa:
  assumes "∀x. Principio x ⟶ Generado x" "¬ Generado x"
  shows "¬ Principio x"
  using assms by blast

section ‹Zenón: la dicotomía (Aristóteles, Física VI 9, 239b11-14)›

(* Para recorrer un estadio hay que recorrer antes la mitad, y antes la mitad de lo que queda, y así sin fin.
   La respuesta moderna: infinitas mitades suman una distancia finita. 1/2 + 1/4 + 1/8 + ... = 1. *)
theorem dicotomia: "(λn. (1/2 :: real) ^ Suc n) sums 1"
proof -
  have "(λn. (1/2 :: real) ^ n) sums (1 / (1 - 1/2))" by (rule geometric_sums) simp
  then have "(λn. (1/2 :: real) * (1/2) ^ n) sums ((1/2) * (1 / (1 - 1/2)))" by (rule sums_mult)
  then show ?thesis by simp
qed

section ‹Lo uno según Aristóteles: lo que separaría dos entes sería el no ser›

(* Aristóteles, Sobre la generación y la corrupción I 8, 325a2-6, lee así a los eléatas: dos entes solo se
   distinguen si entre ellos hay no ser, el vacío; y el vacío no es. Los atomistas aceptan la primera premisa y
   sueltan la segunda. *)
consts Vacio :: "cosa ⇒ bool"
definition P_sin_vacio :: bool where "P_sin_vacio ≡ ∀x. ¬ Vacio x"
definition P_separa :: bool where "P_separa ≡ ∀x y. x ≠ y ∧ ¬ Vacio x ∧ ¬ Vacio y ⟶ (∃z. Vacio z)"

theorem uno_sin_vacio:
  assumes "P_sin_vacio" "P_separa"
  shows "∀x y. ¬ Vacio x ∧ ¬ Vacio y ⟶ x = y"
  using assms unfolding P_sin_vacio_def P_separa_def by blast

(* Los atomistas: dos átomos y el vacío entre ellos. Se cumple «lo que separa es el vacío». *)
lemma muchos_con_vacio:
  fixes V :: "nat ⇒ bool"
  defines "V ≡ λx. x = 2"
  shows "(∀x y. x ≠ y ∧ ¬ V x ∧ ¬ V y ⟶ (∃z. V z)) ∧ ¬ V 0 ∧ ¬ V 1 ∧ V 2"
  unfolding V_def by auto

section ‹Los atomistas: el vacío (Leucipo y Demócrito, según Aristóteles, Metafísica I 4, 985b4-10)›

(* Los atomistas aceptan que nada viene de la nada, pero sueltan «el no ser no es»: el vacío, el no ser, es tanto
   como el ser. Modelo: los lugares son números; el lleno se mueve de un lugar a otro a través del vacío. Hay
   pluralidad (dos átomos) y movimiento, y ningún átomo empieza ni deja de ser. *)
lemma atomistas:
  fixes Lleno :: "nat ⇒ nat ⇒ bool" (* el lugar l está lleno en el instante t *)
  defines "Lleno ≡ λl t. l = t ∨ l = t + 5"
  shows "(∃l t. ¬ Lleno l t) ∧ Lleno 0 0 ∧ ¬ Lleno 0 1 ∧ Lleno 1 1 ∧ (∀t. ∃l1 l2. l1 ≠ l2 ∧ Lleno l1 t ∧ Lleno l2 t)"
proof (intro conjI)
  show "∃l t. ¬ Lleno l t" unfolding Lleno_def by (rule exI [of _ 1], rule exI [of _ 0]) simp
  show "Lleno 0 0" unfolding Lleno_def by simp
  show "¬ Lleno 0 1" unfolding Lleno_def by simp
  show "Lleno 1 1" unfolding Lleno_def by simp
  show "∀t. ∃l1 l2. l1 ≠ l2 ∧ Lleno l1 t ∧ Lleno l2 t"
  proof
    fix t :: nat
    show "∃l1 l2. l1 ≠ l2 ∧ Lleno l1 t ∧ Lleno l2 t" unfolding Lleno_def
      by (rule exI [of _ t], rule exI [of _ "t + 5"]) simp
  qed
qed

end
