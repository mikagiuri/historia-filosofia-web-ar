Theory Timeo_demiurgo

(* El Demiurgo del Timeo (23-10-2026). Sexto mundo de la mesa «Spinoza», a propuesta del profesor.
   Fuentes: Platón, Timeo 27d-29a, 29e-30a, 31a-b, 37d-38b (ed. Burnet); Aristóteles, Sobre el cielo I 10,
   279b17-280a11 (la lectura literal y la «didáctica» de Jenócrates). Traducciones propias. Platón llama a su relato
   eikṑs mŷthos, relato verosímil: la lógica muestra su estructura, no si es verdad. Debate abierto, sin veredicto. *)
theory Timeo_demiurgo
  imports Main
begin

typedecl ente
consts Siempre :: "ente ⇒ bool"          (* lo que siempre es y no deviene *)
consts Deviene :: "ente ⇒ bool"          (* lo que deviene y nunca es del todo *)
consts Visible :: "ente ⇒ bool"          (* visible y tangible: tiene cuerpo *)
consts Causa :: "ente ⇒ ente ⇒ bool"    (* Causa d x: d hace x *)
consts Bello :: "ente ⇒ bool"
consts Bueno :: "ente ⇒ bool"
consts Copia :: "ente ⇒ ente ⇒ bool"    (* Copia x m: x está hecho mirando al modelo m *)
consts cosmos :: ente

section ‹Hay un hacedor y un modelo eterno (27d-29a)›

(* 27d-28a: dos planos, lo que siempre es y lo que deviene; nada está en los dos. *)
definition D_dos_planos :: bool where "D_dos_planos ≡ ∀x. ¬ (Siempre x ∧ Deviene x)"
(* 28a: todo lo que deviene deviene por alguna causa. *)
definition D_causa :: bool where "D_causa ≡ ∀x. Deviene x ⟶ (∃d. Causa d x)"
(* 28b-c: lo visible y tangible deviene; y el cosmos es visible y tangible. *)
definition D_visible :: bool where "D_visible ≡ ∀x. Visible x ⟶ Deviene x"
definition D_cosmos_visible :: bool where "D_cosmos_visible ≡ Visible cosmos"
(* La mesa no tiene nombres propios: habla de lo visible, sea lo que sea. Las mismas premisas, para todo x. *)
definition D_bello :: bool where "D_bello ≡ ∀x. Visible x ⟶ Bello x"
definition D_hacedor_bueno :: bool where "D_hacedor_bueno ≡ ∀d x. Causa d x ∧ Visible x ⟶ Bueno d"
(* 29a: si el cosmos es bello y su hacedor bueno, este miró a un modelo eterno. Son premisas de valor. *)
definition D_bello_cosmos :: bool where "D_bello_cosmos ≡ Bello cosmos"
definition D_hacedor_bueno_cosmos :: bool where "D_hacedor_bueno_cosmos ≡ ∀d. Causa d cosmos ⟶ Bueno d"
definition D_modelo :: bool where
  "D_modelo ≡ ∀d x. Causa d x ∧ Bello x ∧ Bueno d ⟶ (∃m. Siempre m ∧ Copia x m)"

theorem hay_hacedor:
  assumes "D_causa" "D_visible" "D_cosmos_visible"
  shows "∃d. Causa d cosmos"
  using assms unfolding D_causa_def D_visible_def D_cosmos_visible_def by blast

theorem modelo_eterno:
  assumes "D_causa" "D_visible" "D_cosmos_visible" "D_bello_cosmos" "D_hacedor_bueno_cosmos" "D_modelo"
  shows "∃m. Siempre m ∧ Copia cosmos m"
proof -
  obtain d where c: "Causa d cosmos" using hay_hacedor[OF assms(1-3)] by blast
  have "Bueno d" using c assms(5) unfolding D_hacedor_bueno_cosmos_def by blast
  then show ?thesis using c assms(4,6) unfolding D_bello_cosmos_def D_modelo_def by blast
qed

theorem hay_hacedor_mesa:
  assumes "D_causa" "D_visible"
  shows "∀x. Visible x ⟶ (∃d. Causa d x)"
  using assms unfolding D_causa_def D_visible_def by blast

theorem modelo_eterno_mesa:
  assumes "D_causa" "D_visible" "D_bello" "D_hacedor_bueno" "D_modelo"
  shows "∀x. Visible x ⟶ (∃m. Siempre m ∧ Copia x m)"
proof (intro allI impI)
  fix x assume v: "Visible x"
  obtain d where c: "Causa d x" using v assms(1,2) unfolding D_causa_def D_visible_def by blast
  have "Bueno d" using c v assms(4) unfolding D_hacedor_bueno_def by blast
  moreover have "Bello x" using v assms(3) unfolding D_bello_def by blast
  ultimately show "∃m. Siempre m ∧ Copia x m" using c assms(5) unfolding D_modelo_def by blast
qed

(* El modelo no es una copia de lo que deviene: está en el otro plano. *)
theorem el_modelo_no_deviene:
  assumes "D_dos_planos" "Siempre m"
  shows "¬ Deviene m"
  using assms unfolding D_dos_planos_def by blast

(* Sin la premisa del hacedor bueno, nada obliga a un modelo eterno: Platón dice que pensarlo «no es lícito ni
   decirlo» (29a), pero eso ya no es lógica. Modelo: el hacedor no es bueno y el cosmos no copia nada. *)
lemma sin_hacedor_bueno:
  fixes S Dv V B Bu :: "bool ⇒ bool" and C Cp :: "bool ⇒ bool ⇒ bool"
  defines "S ≡ λx. False" and "Dv ≡ λx. x" and "V ≡ λx. x" and "B ≡ λx. True" and "Bu ≡ λx. False"
    and "C ≡ λd x. ¬ d ∧ x" and "Cp ≡ λx m. False"
  shows "(∀x. Dv x ⟶ (∃d. C d x)) ∧ (∀x. V x ⟶ Dv x) ∧ V True ∧ B True
         ∧ (∀d x. C d x ∧ B x ∧ Bu d ⟶ (∃m. S m ∧ Cp x m)) ∧ ¬ (∃m. S m ∧ Cp True m)"
  unfolding S_def Dv_def V_def B_def Bu_def C_def Cp_def by auto

section ‹El cosmos es uno (31a-b)›

(* Si hubiera dos cosmos copiados del mismo modelo, haría falta otro modelo que los abarcara a los dos, y serían
   copia de ese, no del primero. Y un cosmos es copia de un solo modelo, el completo. *)
definition U_dos_copias :: bool where
  "U_dos_copias ≡ ∀c1 c2 m. Copia c1 m ∧ Copia c2 m ∧ c1 ≠ c2 ⟶ (∃m'. Copia c1 m' ∧ m' ≠ m)"
definition U_un_modelo :: bool where "U_un_modelo ≡ ∀c m m'. Copia c m ∧ Copia c m' ⟶ m = m'"

theorem un_solo_cosmos:
  assumes "U_dos_copias" "U_un_modelo" "Copia cosmos m" "Copia c m"
  shows "c = cosmos"
proof (rule ccontr)
  assume "c ≠ cosmos"
  then obtain m' where "Copia c m'" "m' ≠ m" using assms(1,3,4) unfolding U_dos_copias_def by blast
  then show False using assms(2,4) unfolding U_un_modelo_def by blast
qed

theorem un_solo_cosmos_mesa:
  assumes "U_dos_copias" "U_un_modelo"
  shows "∀c1 c2 m. Copia c1 m ∧ Copia c2 m ⟶ c1 = c2"
  using assms unfolding U_dos_copias_def U_un_modelo_def by blast

(* Todo el peso está en U_dos_copias: sin ella, dos cosmos pueden copiar el mismo modelo. *)
lemma sin_dos_copias:
  fixes Cp :: "nat ⇒ nat ⇒ bool"
  defines "Cp ≡ λx m. m = 0 ∧ (x = 1 ∨ x = 2)"
  shows "(∀c m m'. Cp c m ∧ Cp c m' ⟶ m = m') ∧ Cp 1 0 ∧ Cp 2 0"
  unfolding Cp_def by simp

section ‹¿Empezó el cosmos? (28b: gégonen, «ha llegado a ser»)›

typedecl instante
consts Antes :: "instante ⇒ instante ⇒ bool"
consts Existe :: "ente ⇒ instante ⇒ bool"
definition comienza :: "ente ⇒ bool" where
  "comienza x ≡ ∃t. Existe x t ∧ (∀s. Antes s t ⟶ ¬ Existe x s)"

(* La lectura literal, la de Aristóteles (Sobre el cielo I 10): lo hecho empieza a ser. *)
definition L_literal :: bool where "L_literal ≡ ∀d x. Causa d x ⟶ comienza x"

theorem lectura_literal:
  assumes "L_literal" "D_causa" "D_visible" "D_cosmos_visible"
  shows "comienza cosmos"
  using hay_hacedor[OF assms(2-4)] assms(1) unfolding L_literal_def by blast

theorem lectura_literal_mesa:
  assumes "L_literal" "D_causa" "D_visible"
  shows "∀x. Visible x ⟶ comienza x"
  using assms unfolding L_literal_def D_causa_def D_visible_def by blast

(* Con la lectura literal, nada hecho deja de comenzar: es la premisa misma. *)
lemma literal_comienza:
  assumes "L_literal" "Causa d x"
  shows "comienza x"
  using assms unfolding L_literal_def by blast

(* La lectura de dependencia (Jenócrates y los neoplatónicos): «hecho» quiere decir «causado», no «empezado».
   Modelo: el cosmos es causado y existe en todos los instantes, que son los enteros, sin primero. *)
lemma lectura_dependencia:
  fixes C :: "nat ⇒ nat ⇒ bool" and E :: "nat ⇒ int ⇒ bool"
  defines "C ≡ λd x. d = 0 ∧ x = 1" and "E ≡ λx t. True"
  shows "C 0 1 ∧ ¬ (∃t. E 1 t ∧ (∀s. s < t ⟶ ¬ E 1 s))"
proof (intro conjI)
  show "C 0 1" unfolding C_def by simp
  show "¬ (∃t. E 1 t ∧ (∀s. s < t ⟶ ¬ E 1 s))"
  proof
    assume "∃t. E 1 t ∧ (∀s. s < t ⟶ ¬ E 1 s)"
    then obtain t :: int where "∀s. s < t ⟶ ¬ E 1 s" by blast
    then have "¬ E 1 (t - 1)" by simp
    then show False unfolding E_def by simp
  qed
qed

section ‹El tiempo nace con el cosmos (37d-38b)›

(* El tiempo es «imagen móvil de la eternidad» y nace con el cielo. Modelo: los instantes son los naturales y el
   cosmos existe en todos. Tiene comienzo (el 0) y, sin embargo, no hay ningún instante en que no existiera: no hay
   un «antes» vacío. Es soltar la premisa A1_tiempo_sin_principio de la primera antinomia. *)
lemma tiempo_con_el_cosmos:
  fixes E :: "nat ⇒ bool"
  defines "E ≡ λt. True"
  shows "(∃t0. E t0 ∧ (∀s. s < t0 ⟶ ¬ E s)) ∧ (∀t. E t) ∧ ¬ (∀t::nat. ∃s. s < t)"
proof (intro conjI)
  show "∃t0. E t0 ∧ (∀s. s < t0 ⟶ ¬ E s)" unfolding E_def by (rule exI [of _ 0]) simp
  show "∀t. E t" unfolding E_def by simp
  show "¬ (∀t::nat. ∃s. s < t)"
  proof
    assume "∀t::nat. ∃s. s < t"
    then obtain s :: nat where "s < 0" by blast
    then show False by simp
  qed
qed
