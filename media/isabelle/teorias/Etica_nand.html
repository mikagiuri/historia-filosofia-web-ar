Theory Etica_nand

theory Etica_nand
imports Main
begin

text ‹La única conectiva: NAND, la barra de Sheffer. Se escribe como función prefija
  ‹nand a b› (para evitar ambigüedades de precedencia; en la literatura es la barra
  $a \<uparrow> b$). Todo lo demás se define a partir de ella. Es el mismo gesto que en la
  metafísica: de cuatro conectivas (y antes, de veinte relaciones) a UNA sola.›

definition nand :: "bool ⇒ bool ⇒ bool" where
  "nand a b ≡ ¬ (a ∧ b)"

text ‹Las tres conectivas básicas se reducen a NAND:›
lemma neg_nand:  "((¬ a) = nand a a)"  by (auto simp add: nand_def)
lemma conj_nand: "((a ∧ b) = nand (nand a b) (nand a b))"  by (auto simp add: nand_def)
lemma disj_nand: "((a ∨ b) = nand (nand a a) (nand b b))"  by (auto simp add: nand_def)

text ‹Y las otras dos, por composición:›
lemma imp_nand:  "((a ⟶ b) = nand a (nand b b))"  by (auto simp add: nand_def)
(* el bicondicional ya se había reducido a ¬/∨/∧, así que compone *)

text ‹Abreviaturas en NAND puro (no = negación, y = conjunción, ou = disyunción,
  en español):›
abbreviation no :: "bool ⇒ bool" where "no a ≡ nand a a"
abbreviation y  :: "bool ⇒ bool ⇒ bool" where "y a b ≡ nand (nand a b) (nand a b)"
abbreviation ou :: "bool ⇒ bool ⇒ bool" where "ou a b ≡ nand (nand a a) (nand b b)"

consts
  Adecuacion :: bool
  Actividad :: bool
  Virtud :: bool
  Beatitud :: bool
  Eficaz :: bool
  Conato :: bool

axiomatization where
  A_adecuacion_actividad: "(Actividad ∧ Adecuacion) ∨ (¬ Actividad ∧ ¬ Adecuacion)" and
  A_virtud_actividad:     "(Virtud ∧ Actividad) ∨ (¬ Virtud ∧ ¬ Actividad)" and
  A_beatitud_virtud:      "(Beatitud ∧ Virtud) ∨ (¬ Beatitud ∧ ¬ Virtud)" and
  A_conato_es_potencia:   "(Conato ∧ Eficaz) ∨ (¬ Conato ∧ ¬ Eficaz)"

text ‹La espina del libro —adecuación = actividad = virtud = beatitud— escrita
  ENTERAMENTE en NAND (solo ‹nand›, sin ¬ ni ∨ ni ∧), y demostrada.›
theorem libertad_es_beatitud_nand:
  "ou (y Adecuacion Beatitud) (y (no Adecuacion) (no Beatitud))"
  unfolding nand_def
  using A_adecuacion_actividad A_virtud_actividad A_beatitud_virtud
  by blast

text ‹Y la cadena de la potencia —conato = eficacia— también en NAND:›
theorem conato_es_potencia_nand:
  "ou (y Conato Eficaz) (y (no Conato) (no Eficaz))"
  using A_conato_es_potencia unfolding nand_def by blast

text ‹La fórmula gigante: el bicondicional «adecuación ⟷ beatitud» expandido
  íntegro a pura NAND (23 nand, sin ninguna otra conectiva), y demostrado igual al
  bicondicional.›
lemma gigante:
  "nand (nand (nand (nand Adecuacion Beatitud) (nand Adecuacion Beatitud))
               (nand (nand Adecuacion Beatitud) (nand Adecuacion Beatitud)))
        (nand (nand (nand (nand Adecuacion Adecuacion) (nand Beatitud Beatitud))
                     (nand (nand Adecuacion Adecuacion) (nand Beatitud Beatitud)))
              (nand (nand (nand Adecuacion Adecuacion) (nand Beatitud Beatitud))
                     (nand (nand Adecuacion Adecuacion) (nand Beatitud Beatitud))))
   = (Adecuacion ⟷ Beatitud)"
  by (auto simp add: nand_def)

end
