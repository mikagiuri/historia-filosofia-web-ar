Theory Spinoza_identidad_minimal

(* Versión NUEVA: núcleo mínimo con la lectura de IDENTIDAD,
   y ahora siguiendo el ORDEN de exposición de Spinoza (def. 1, 3, 5 → ax. 1 → P15).

   Relación con la Parte I de Spinoza:
   - def. 1 (causa sui) y def. 3 (in se)  →  se unifican en "depende x x"
   - def. 5 (modo = in alio)              →  modo
   - ax. 1 (todo es en sí o en otro)      →  A1_exhaustividad
   - P14 (no hay sustancia fuera de Dios) →  A_unicidad (ASUMIDO, no derivado)
   - P15 (todo es en Dios)                →  ahora DEMOSTRABLE
   Las P2-P6, P7/P11, P13 quedan absorbidas por A_unicidad y A_Dios.
   Las P3-P4, P8-P10, P12 piden vocabulario (común, partes, infinito) que aquí se omite. *)

theory Spinoza_identidad_minimal
  imports Main
begin

section ‹Vocabulario: un solo primitivo›

typedecl cosa
typedecl atributo

consts
  depende  :: "cosa ⇒ cosa ⇒ bool"      ― ‹fundamentación: ser-en = concebirse-por = ser-causado-por›
  expresa  :: "cosa ⇒ atributo ⇒ bool"  ― ‹many-to-many: NO individualiza›
  Extension   :: atributo
  Pensamiento :: atributo
  Dios        :: cosa

section ‹Definiciones (orden de Spinoza: def. 1, 3, 5)›

text ‹Def. 1 (causa sui) = lo que es en sí. En este vocabulario, «ser en sí» y «ser
  causa de sí» son literalmente lo mismo: depender de sí mismo.›
definition causa_sui :: "cosa ⇒ bool" where
  "causa_sui x ≡ depende x x"

text ‹Def. 5 (modo) = lo que es en otra cosa, y no en sí. La cláusula «y no en sí»
  es la exclusividad de Spinoza entre «in se» y «in alio»: hace que un modo no pueda ser
  causa de sí, y por tanto no sea eterno (véase P24).›
definition modo :: "cosa ⇒ bool" where
  "modo x ≡ ∃y. y ≠ x ∧ depende x y ∧ ¬ depende x x"

text ‹Def. 3 (sustancia) = lo que es en sí y no en otra cosa.›
definition sustancia :: "cosa ⇒ bool" where
  "sustancia x ≡ depende x x ∧ (∀y. depende x y ⟶ y = x)"

text ‹La causa se define a partir de la fundamentación (absorbe el ax. 4 de Spinoza).›
definition causa :: "cosa ⇒ cosa ⇒ bool" where
  "causa x y ≡ depende y x"

section ‹Axiomas›

text ‹Ax. 1 de Spinoza: todo es en sí (sustancia) o en otro (modo).›
definition A1_exhaustividad :: bool where
  "A1_exhaustividad ≡ ∀x. sustancia x ∨ modo x"

text ‹Def. 6 (Dios): la sustancia que expresa todos los atributos.›
definition A_Dios :: bool where
  "A_Dios ≡ sustancia Dios ∧ (∀a. expresa Dios a)"

text ‹P14 (no hay sustancia fuera de Dios): aquí ASUMIDO, no derivado.›
definition A_unicidad :: bool where
  "A_unicidad ≡ ∀s. sustancia s ⟶ s = Dios"

text ‹Todo modo se funda en Dios (la mitad «modos» de P15).›
definition A_fundamento :: bool where
  "A_fundamento ≡ ∀m. modo m ⟶ depende m Dios"

text ‹Extensión y Pensamiento son atributos distintos (pero de la única sustancia).›
definition A_atributos_distintos :: bool where
  "A_atributos_distintos ≡ Extension ≠ Pensamiento"

text ‹La lectura de identidad: todo modo expresa AMBOS atributos (un humano = un modo).›
definition A_identidad :: bool where
  "A_identidad ≡ ∀m. modo m ⟶ expresa m Extension ∧ expresa m Pensamiento"

section ‹Proposiciones en el orden de Spinoza›

text ‹P15: lo que es, es en Dios, y nada puede ser ni concebirse sin Dios.
  En este vocabulario: toda cosa depende de Dios (también Dios, que es causa sui).›
theorem P15:
  assumes "A1_exhaustividad" "A_unicidad" "A_fundamento" "A_Dios"
  shows "∀x. depende x Dios"
proof
  fix x
  have ex: "sustancia x ∨ modo x" using assms(1) unfolding A1_exhaustividad_def by blast
  show "depende x Dios"
  proof (cases "sustancia x")
    case True
    then have "x = Dios" using assms(2) unfolding A_unicidad_def by blast
    moreover have "depende Dios Dios" using assms(4) unfolding A_Dios_def sustancia_def by blast
    ultimately show ?thesis by simp
  next
    case False
    then have "modo x" using ex by blast
    then show ?thesis using assms(3) unfolding A_fundamento_def by blast
  qed
qed

section ‹Simetría de los atributos: idea y objeto, mente y cuerpo›

text ‹«La mente es la idea del cuerpo» y «el cuerpo es el objeto de la mente» son la
  MISMA relación invertida: un único modo expresado bajo dos atributos, sin que ninguno
  tenga prioridad sobre el otro. En esta lectura no hay una relación asimétrica «IdeaDe»
  que vaya de la idea al objeto: la correspondencia ES la identidad. Formalmente, para
  todo modo, ser-extenso y ser-pensante son equivalentes: no cabe uno sin el otro.›

theorem simetria_atributos:
  assumes "A_identidad" "modo m"
  shows "expresa m Extension ⟷ expresa m Pensamiento"
  using assms unfolding A_identidad_def by blast

text ‹Consecuencia: no puede existir un modo que sea cuerpo pero no mente...›
lemma no_asimetria_cuerpo_sin_mente:
  "A_identidad ∧ (∃m. modo m ∧ expresa m Extension ∧ ¬ expresa m Pensamiento)"
  nitpick [satisfy, timeout = 120]
  oops

text ‹... ni un modo que sea mente pero no cuerpo. La simetría queda garantizada.›
lemma no_asimetria_mente_sin_cuerpo:
  "A_identidad ∧ (∃m. modo m ∧ expresa m Pensamiento ∧ ¬ expresa m Extension)"
  nitpick [satisfy, timeout = 120]
  oops

section ‹La gradación del ser y del concebir›

consts
  MasRealidad   :: "cosa ⇒ cosa ⇒ bool"  ― ‹x tiene más potencia de ser que y›
  MasConcebible :: "cosa ⇒ cosa ⇒ bool"  ― ‹x es más inteligible que y›
  Concebible    :: "cosa ⇒ bool"           ― ‹x es concebible›
  Inconsistente :: "cosa ⇒ bool"           ― ‹x es racionalmente inconsistente›

text ‹El principio (P9 + paralelismo, en grados): el orden del ser y el orden del
  concebir son el mismo. A más potencia de ser, más potencia de concepto.›
definition A_gradacion :: bool where
  "A_gradacion ≡ ∀x y. MasRealidad x y ⟷ MasConcebible x y"

text ‹El corte lógico: lo racionalmente inconsistente no tiene ser, luego no es
  concebible (su concebibilidad es cero, no mínima).›
definition A_cero :: bool where
  "A_cero ≡ ∀x. Inconsistente x ⟶ ¬ Concebible x"

text ‹Lo concebible es lo que está bajo el Pensamiento: la concebibilidad ES el
  aspecto mental de la cosa.›
definition A_concebible_es_pensamiento :: bool where
  "A_concebible_es_pensamiento ≡ ∀x. Concebible x ⟷ expresa x Pensamiento"

text ‹Dios es lo más real (ens realissimum).›
definition A_Dios_realissimo :: bool where
  "A_Dios_realissimo ≡ ∀m. modo m ⟶ MasRealidad Dios m"

text ‹Consecuencia: ningún modo es inconsistente (todo modo es real y, por tanto,
  concebible). Lo inconsistente solo habita fuera del ser.›
theorem modo_no_inconsistente:
  assumes "A_cero" "A_concebible_es_pensamiento" "A_identidad" "modo m"
  shows "¬ Inconsistente m"
  using assms unfolding A_cero_def A_concebible_es_pensamiento_def A_identidad_def by blast

text ‹La identidad con atributos distintos: un modo expresa DOS atributos genuinamente
  distintos (no uno solo). Es la versión positiva de lo que la lectura antigua niega.›
theorem modo_expresa_atributos_distintos:
  assumes "A_identidad" "A_atributos_distintos" "modo m"
  shows "expresa m Extension ∧ expresa m Pensamiento ∧ Extension ≠ Pensamiento"
  using assms unfolding A_identidad_def A_atributos_distintos_def by blast

text ‹De la gradación: lo más real es lo más concebible. Dios, que es lo más real,
  es lo más concebible (la cúspide del orden del ser y del concebir).›
theorem Dios_lo_mas_concebible:
  assumes "A_gradacion" "A_Dios_realissimo"
  shows "∀m. modo m ⟶ MasConcebible Dios m"
  using assms unfolding A_gradacion_def A_Dios_realissimo_def by blast

text ‹Todo el sistema junto con la gradación: ¿sigue siendo consistente?›
lemma consistencia_gradacion:
  "A1_exhaustividad ∧ A_Dios ∧ A_unicidad ∧ A_fundamento ∧ A_atributos_distintos ∧ A_identidad
   ∧ A_gradacion ∧ A_cero ∧ A_concebible_es_pensamiento ∧ A_Dios_realissimo
   ∧ (∃m1 m2. modo m1 ∧ modo m2 ∧ m1 ≠ m2)"
  nitpick [satisfy, timeout = 120]
  oops

text ‹El cero es un corte real: nada puede ser a la vez inconsistente y concebible.›
lemma cero_imposible:
  "A_cero ∧ (∃x. Inconsistente x ∧ Concebible x)"
  nitpick [satisfy, timeout = 120]
  oops

text ‹La jerarquía: puede haber dos modos, uno con más realidad y más concebibilidad
  que el otro (la piedra y el humano). Ambos están «bajo el Pensamiento» (A_identidad),
  pero en grados distintos: la piedra no «piensa» igual que el humano.›
lemma jerarquia_modos:
  "A_gradacion ∧ (∃m1 m2. modo m1 ∧ modo m2 ∧ m1 ≠ m2
     ∧ MasRealidad m2 m1 ∧ MasConcebible m2 m1)"
  nitpick [satisfy, timeout = 120]
  oops

section ‹P16-P18: Dios como causa›

text ‹P16 cor. 1: Dios es causa eficiente de todas las cosas. En este vocabulario es
  inmediato de P15: «ser en Dios» ES «ser causado por Dios» (definición de causa). La
  premisa oculta H_inherencia_es_causalidad que Spinoza necesitaba aquí es gratis.›
theorem P16_cor1:
  assumes "A1_exhaustividad" "A_unicidad" "A_fundamento" "A_Dios"
  shows "∀x. causa Dios x"
  using P15[OF assms] unfolding causa_def by blast

text ‹P17: Dios obra por las solas leyes de su naturaleza, sin ser coaccionado por
  nadie. Formalmente: nada distinto de Dios causa a Dios.›
theorem P17:
  assumes "A_Dios"
  shows "∀y. causa y Dios ⟶ y = Dios"
  using assms unfolding A_Dios_def sustancia_def causa_def by blast

text ‹P18: Dios es causa inmanente, y no transitiva, de todas las cosas. La immanencia
  es automática: como causa = fundamentación, el efecto siempre «es en» su causa.›
definition causa_inmanente :: "cosa ⇒ cosa ⇒ bool" where
  "causa_inmanente y x ≡ causa y x ∧ depende x y"

definition causa_transitiva :: "cosa ⇒ cosa ⇒ bool" where
  "causa_transitiva y x ≡ causa y x ∧ ¬ depende x y"

theorem P18:
  assumes "A1_exhaustividad" "A_unicidad" "A_fundamento" "A_Dios"
  shows "∀x. causa_inmanente Dios x ∧ ¬ causa_transitiva Dios x"
  using P15[OF assms] unfolding causa_def causa_inmanente_def causa_transitiva_def by blast

section ‹P19 y P24: la eternidad›

text ‹Def. 8 (eternidad): la existencia misma, en cuanto se concibe que se sigue
  necesariamente de la sola definición de la cosa eterna. En este vocabulario, «eterno»
  es «causa sui»: existir por su propia naturaleza, depender de sí mismo. Las defs. 1
  (causa sui), 3 (in se) y 8 (eterno) se unifican en una sola: $depende x x$.›
definition eterno :: "cosa ⇒ bool" where
  "eterno x ≡ depende x x"

text ‹Un atributo es eterno si lo expresa una sustancia (que es, por definición, eterna).›
definition atributo_eterno :: "atributo ⇒ bool" where
  "atributo_eterno a ≡ ∃s. sustancia s ∧ expresa s a"

text ‹P19: Dios, o sea, todos los atributos de Dios, son eternos.›
theorem P19:
  assumes "A_Dios"
  shows "eterno Dios ∧ (∀a. atributo_eterno a)"
proof -
  have e: "eterno Dios" using assms unfolding A_Dios_def eterno_def sustancia_def by blast
  moreover have "∀a. atributo_eterno a"
    using assms unfolding A_Dios_def atributo_eterno_def by blast
  ultimately show ?thesis by blast
qed

text ‹P24: la esencia de las cosas producidas por Dios no implica la existencia.
  O sea: ningún modo es eterno; su ser depende de otro, no de sí mismo.›
theorem P24:
  assumes "modo m"
  shows "¬ eterno m"
  using assms unfolding modo_def eterno_def by blast

section ‹P21-P23: la doble eternidad y los modos infinitos›

consts
  SigueDeAtributoAbsoluto :: "cosa ⇒ atributo ⇒ bool"  ― ‹x se sigue de la naturaleza absoluta de a›
  SigueDeModificacion     :: "cosa ⇒ cosa ⇒ bool"      ― ‹x se sigue del atributo modificado por y›

text ‹La segunda eternidad: eterno POR EL ATRIBUTO (se sigue, inmediata o mediatamente,
  de un atributo eterno). Es distinta de la eternidad por la esencia (= causa sui), que
  solo Dios tiene. Es la «doble eternidad» de Spinoza.›
definition eterno_por_atributo :: "cosa ⇒ bool" where
  "eterno_por_atributo x ≡ (∃a. SigueDeAtributoAbsoluto x a)
     ∨ (∃y a. SigueDeModificacion x y ∧ SigueDeAtributoAbsoluto y a)"

text ‹Lo que se sigue de la naturaleza absoluta de un atributo es un modo cuya única
  causa es Dios (se sigue sin intermediarios).›
definition H_absoluto :: bool where
  "H_absoluto ≡ ∀x a. SigueDeAtributoAbsoluto x a ⟶ modo x ∧ (∀z. causa z x ⟶ z = Dios)"

text ‹Todo modo se sigue de Dios de una de las dos maneras: absoluta o mediata.›
definition H_dos_maneras :: bool where
  "H_dos_maneras ≡ ∀m. modo m ⟶ (∃a. SigueDeAtributoAbsoluto m a) ∨ (∃y. SigueDeModificacion m y)"

text ‹P21: lo que se sigue de la naturaleza absoluta de un atributo es eterno POR EL
  ATRIBUTO, pero NO por su esencia. Es la doble eternidad: los modos infinitos son
  eternos sin ser causa de sí.›
theorem P21:
  assumes "H_absoluto" "SigueDeAtributoAbsoluto x a"
  shows "eterno_por_atributo x ∧ ¬ eterno x"
proof
  show "eterno_por_atributo x" using assms(2) unfolding eterno_por_atributo_def by blast
  show "¬ eterno x"
  proof (rule P24)
    show "modo x" using assms unfolding H_absoluto_def by blast
  qed
qed

text ‹¿Es consistente que exista un modo que se sigue de la naturaleza absoluta
  (un modo infinito inmediato)?›
lemma consistencia_modo_infinito:
  "H_absoluto ∧ H_dos_maneras ∧ (∃x a. SigueDeAtributoAbsoluto x a)"
  nitpick [satisfy, timeout = 120]
  oops

section ‹P22-P23: el modo infinito mediato y lo finito›

consts
  Limita :: "cosa ⇒ cosa ⇒ bool"  ― ‹Limita y x: y limita a x›

text ‹Def. 2 (finito): es finito lo que puede ser limitado por otro de la misma
  naturaleza. Aquí «infinito» es, simplemente, no-finito.›
definition finita_modo :: "cosa ⇒ bool" where
  "finita_modo x ≡ ∃y. y ≠ x ∧ Limita y x"

text ‹Limitar es determinar (causar): lo que limita a una cosa la causa.›
definition H_limitar_es_determinar :: bool where
  "H_limitar_es_determinar ≡ ∀x y. Limita y x ⟶ causa y x"

text ‹Lo que limita a otra cosa es un modo finito.›
definition H_lo_que_limita_es_modo_finito :: bool where
  "H_lo_que_limita_es_modo_finito ≡ ∀x y. Limita y x ∧ y ≠ x ⟶ modo y ∧ finita_modo y"

text ‹Lo que se sigue de un atributo modificado por y tiene a y por causa (y solo a
  Dios o a y).›
definition H_mediato :: bool where
  "H_mediato ≡ ∀x y. SigueDeModificacion x y ⟶ modo y ∧ causa y x ∧ (∀z. causa z x ⟶ z = Dios ∨ z = y)"

text ‹De lo finito se sigue lo finito.›
definition H_de_lo_finito_lo_finito :: bool where
  "H_de_lo_finito_lo_finito ≡ ∀m y. SigueDeModificacion m y ∧ finita_modo y ⟶ finita_modo m"

lemma modo_no_sustancia:
  assumes "modo x"
  shows "¬ sustancia x"
  using assms unfolding modo_def sustancia_def by blast

text ‹P22: lo que se sigue de un atributo modificado por una modificación infinita es
  también infinito (el modo infinito MEDIATO).›
theorem P22:
  assumes "H_limitar_es_determinar" "H_lo_que_limita_es_modo_finito" "H_mediato" "A_Dios"
          "SigueDeModificacion x y" "¬ finita_modo y"
  shows "¬ finita_modo x"
proof
  assume fx: "finita_modo x"
  then obtain w where "w ≠ x" and "Limita w x" unfolding finita_modo_def by blast
  have cwx: "causa w x" using ‹Limita w x› assms(1) unfolding H_limitar_es_determinar_def by blast
  have mw: "modo w" and fw: "finita_modo w"
    using ‹w ≠ x› ‹Limita w x› assms(2) unfolding H_lo_que_limita_es_modo_finito_def by blast+
  have wy: "w = Dios ∨ w = y" using cwx assms(3) assms(5) unfolding H_mediato_def by blast
  have "w ≠ Dios"
  proof
    assume "w = Dios"
    then have "sustancia w" using assms(4) unfolding A_Dios_def by blast
    then show False using mw modo_no_sustancia [of w] by blast
  qed
  then have "w = y" using wy by blast
  then have "finita_modo y" using fw by simp
  then show False using assms(6) by blast
qed

text ‹P23: todo modo infinito se sigue de la naturaleza absoluta de un atributo, o de un
  atributo modificado por una modificación infinita.›
theorem P23:
  assumes "H_dos_maneras" "H_de_lo_finito_lo_finito" "modo m" "¬ finita_modo m"
  shows "(∃a. SigueDeAtributoAbsoluto m a) ∨ (∃y. SigueDeModificacion m y ∧ ¬ finita_modo y)"
proof (cases "∃a. SigueDeAtributoAbsoluto m a")
  case True
  then show ?thesis by blast
next
  case False
  then have "∃y. SigueDeModificacion m y"
    using assms(1) assms(3) unfolding H_dos_maneras_def by blast
  then obtain y where sy: "SigueDeModificacion m y" by blast
  have "¬ finita_modo y"
  proof
    assume "finita_modo y"
    then have "finita_modo m" using sy assms(2) unfolding H_de_lo_finito_lo_finito_def by blast
    then show False using assms(4) by blast
  qed
  then show ?thesis using sy by blast
qed

text ‹¿Es consistente un modo infinito mediato (que se sigue de una modificación infinita)?›
lemma consistencia_modo_infinito_mediato:
  "H_mediato ∧ H_de_lo_finito_lo_finito ∧ H_dos_maneras ∧ H_limitar_es_determinar ∧ H_lo_que_limita_es_modo_finito
   ∧ (∃x y a. SigueDeModificacion x y ∧ SigueDeAtributoAbsoluto y a ∧ ¬ finita_modo y)"
  nitpick [satisfy, timeout = 120]
  oops

section ‹P28: la cadena de causas finitas›

text ‹P28: toda cosa finita es determinada a existir y obrar por otra cosa finita, y
  esta a su vez por otra, y así al infinito. Aquí, la pieza central: todo modo finito
  tiene por causa a otro modo finito. Es la cadena HORIZONTAL de causas, que se apoya
  verticalmente en Dios (P15) pero no arranca de ningún modo finito primero.›
theorem P28:
  assumes "H_limitar_es_determinar" "H_lo_que_limita_es_modo_finito" "finita_modo m"
  shows "∃y. y ≠ m ∧ modo y ∧ finita_modo y ∧ causa y m"
proof -
  obtain y where "y ≠ m" and "Limita y m" using assms(3) unfolding finita_modo_def by blast
  moreover have "causa y m" using ‹Limita y m› assms(1) unfolding H_limitar_es_determinar_def by blast
  moreover have "modo y ∧ finita_modo y"
    using ‹y ≠ m› ‹Limita y m› assms(2) unfolding H_lo_que_limita_es_modo_finito_def by blast
  ultimately show ?thesis by blast
qed

text ‹¿Es consistente que haya un modo finito (limitado por otro)?›
lemma consistencia_modo_finito:
  "H_limitar_es_determinar ∧ H_lo_que_limita_es_modo_finito ∧ (∃m. finita_modo m)"
  nitpick [satisfy, timeout = 120]
  oops

section ‹P29, P32, P33: lo modal (necesidad y libertad)›

consts
  ConcebibleComoNoExistente :: "cosa ⇒ bool"  ― ‹x puede concebirse como no existente›
  Voluntad :: cosa

text ‹Lo contingente es lo que puede concebirse como no existente: ni necesario ni
  imposible, «podría ser o no ser».›
definition Contingente :: "cosa ⇒ bool" where
  "Contingente x ≡ ConcebibleComoNoExistente x"

text ‹A3 de Spinoza: de una causa determinada se sigue necesariamente el efecto; lo
  causado no puede concebirse como no existente.›
definition A3 :: bool where
  "A3 ≡ ∀x y. causa y x ⟶ ¬ ConcebibleComoNoExistente x"

text ‹Def. 7 (libre): lo que se determina a obrar por sí solo. Como «eterno» y «causa
  sui», en este vocabulario «libre» es «depender de sí mismo».›
definition Libre :: "cosa ⇒ bool" where
  "Libre x ≡ eterno x"

text ‹P29: nada es contingente; todo es determinado por la necesidad de la naturaleza
  divina. Aquí: todo tiene causa (P16 cor. 1), y lo causado es necesario (A3).›
theorem P29:
  assumes "A3" "A1_exhaustividad" "A_unicidad" "A_fundamento" "A_Dios"
  shows "∀x. ¬ Contingente x"
proof
  fix x
  have "causa Dios x" using P16_cor1[OF assms(2) assms(3) assms(4) assms(5)] by blast
  then have "¬ ConcebibleComoNoExistente x" using assms(1) unfolding A3_def by blast
  then show "¬ Contingente x" unfolding Contingente_def by blast
qed

text ‹P32: la voluntad no puede llamarse causa libre, sino solo necesaria. La voluntad
  es un modo; los modos no son causa de sí (P24); luego la voluntad no es libre.›
theorem P32:
  assumes "modo Voluntad"
  shows "¬ Libre Voluntad"
  using assms P24 unfolding Libre_def by blast

consts
  persona :: cosa
  roca :: cosa

text ‹Ser causa y ser libre son independientes: la persona causa la roca sin ser libre
  de haberla lanzado. Devolverá MODELO: causar no exige ser causa de sí. En la roca la
  persona es causa verdadera (P28 horizontal) sin ser causa libre (P32).›
lemma causa_sin_libertad:
