Theory Propiedades_relaciones

theory Propiedades_relaciones
  imports Main
begin

(* Mapa de propiedades de las relaciones (22-10-2026), para la versión «por capas» de la mesa del Laboratorio:
   qué propiedad de una relación se sigue de cuáles, con prueba, y cuáles no, con contraejemplo.
   Vale para cualquier relación R (Antes, Crea, Mueve, Causa, MasPerfecto…). Mismos nombres que la mesa web.
   Pruebas escritas paso a paso, sin auto ni presburger: en Cinco_vias.thy una de esas búsquedas no terminaba. *)

definition irreflexiva :: "('a ⇒ 'a ⇒ bool) ⇒ bool" where "irreflexiva R ≡ ∀x. ¬ R x x"
definition transitiva :: "('a ⇒ 'a ⇒ bool) ⇒ bool" where "transitiva R ≡ ∀x y z. R x y ∧ R y z ⟶ R x z"
definition asimetrica :: "('a ⇒ 'a ⇒ bool) ⇒ bool" where "asimetrica R ≡ ∀x y. R x y ⟶ ¬ R y x"
(* Sin regreso al infinito: no hay cadena infinita x1 ← x2 ← x3 ← …; la relación está bien fundada. *)
definition sin_regreso :: "('a ⇒ 'a ⇒ bool) ⇒ bool" where "sin_regreso R ≡ wf {(y, x). R y x}"
definition total :: "('a ⇒ 'a ⇒ bool) ⇒ bool" where "total R ≡ ∀x y. x ≠ y ⟶ R x y ∨ R y x"

(* ---------- Lo que se sigue ---------- *)

(* M1. Irreflexiva y transitiva: asimétrica. *)
lemma M1_irreflexiva_transitiva_asimetrica:
  assumes "irreflexiva R" "transitiva R"
  shows "asimetrica R"
  unfolding asimetrica_def
proof (intro allI impI notI)
  fix x y assume "R x y" "R y x"
  then have "R x x" using assms(2) unfolding transitiva_def by blast
  then show False using assms(1) unfolding irreflexiva_def by simp
qed

(* M2. Asimétrica: irreflexiva. *)
lemma M2_asimetrica_irreflexiva:
  assumes "asimetrica R"
  shows "irreflexiva R"
  unfolding irreflexiva_def
proof (intro allI notI)
  fix x assume "R x x"
  then show False using assms unfolding asimetrica_def by blast
qed

(* M3. Sin regreso al infinito: asimétrica (y por M2, irreflexiva). *)
lemma M3_sin_regreso_asimetrica:
  assumes "sin_regreso R"
  shows "asimetrica R"
  unfolding asimetrica_def
proof (intro allI impI notI)
  fix x y assume xy: "R x y" and yx: "R y x"
  let ?r = "{(y, x). R y x}"
  have w: "wf ?r" using assms unfolding sin_regreso_def .
  have "(x, y) ∈ ?r⇧+" using xy by (intro r_into_trancl) simp
  moreover have "(y, x) ∈ ?r⇧+" using yx by (intro r_into_trancl) simp
  ultimately have "(x, x) ∈ ?r⇧+" by (rule trancl_trans)
  moreover have "(x, x) ∉ ?r⇧+" using wf_trancl [OF w] by (rule wf_not_refl)
  ultimately show False by contradiction
qed

(* M4. Sin regreso al infinito, si hay alguna flecha, hay un primero: algo que tiene flechas salientes y ninguna entrante. *)
lemma M4_sin_regreso_hay_primero:
  assumes "sin_regreso R" "R a b"
  shows "∃x. ¬ (∃y. R y x) ∧ (∃z. R x z)"
proof -
  let ?r = "{(y, x). R y x}" and ?S = "{x. ∃z. R x z}"
  have "a ∈ ?S" using assms(2) by blast
  have "∃z∈?S. ∀y. (y, z) ∈ ?r ⟶ y ∉ ?S"
    using wf_eq_minimal[THEN iffD1, OF assms(1)[unfolded sin_regreso_def], rule_format, OF ‹a ∈ ?S›] .
  then obtain x where "x ∈ ?S" and min: "∀y. (y, x) ∈ ?r ⟶ y ∉ ?S" by blast
  have "¬ (∃y. R y x)"
  proof
    assume "∃y. R y x"
    then obtain y where "R y x" by blast
    then have "y ∈ ?S" by blast
    moreover have "(y, x) ∈ ?r" using ‹R y x› by simp
    ultimately show False using min by blast
  qed
  then show ?thesis using ‹x ∈ ?S› by blast
qed

(* M5. En un mundo finito, irreflexiva y transitiva: sin regreso al infinito. *)
lemma M5_finito_sin_regreso:
  assumes fin: "finite (UNIV :: 'a set)" and irr: "irreflexiva (R :: 'a ⇒ 'a ⇒ bool)" and tr: "transitiva R"
  shows "sin_regreso R"
proof -
  let ?r = "{(y, x). R y x}"
  have cl: "(a, b) ∈ ?r⇧+ ⟹ R a b" for a b
  proof (induction rule: trancl_induct)
    case (base y) then show ?case by simp
  next
    case (step y z) then show ?case using tr unfolding transitiva_def by blast
  qed
  have "acyclic ?r" unfolding acyclic_def
  proof (intro allI notI)
    fix x assume "(x, x) ∈ ?r⇧+"
    then have "R x x" by (rule cl)
    then show False using irr unfolding irreflexiva_def by simp
  qed
  moreover have "finite ?r"
  proof -
    have "finite (UNIV :: ('a × 'a) set)" using fin by (simp add: finite_Prod_UNIV)
    then show ?thesis by (rule finite_subset[rotated]) simp
  qed
  ultimately show ?thesis unfolding sin_regreso_def by (simp add: finite_acyclic_wf)
qed

(* M6. En un mundo finito, un orden estricto y total tiene máximo: algo que supera a todo lo demás. *)
lemma M6_finito_total_maximo:
  assumes fin: "finite (UNIV :: 'a set)" and irr: "irreflexiva (R :: 'a ⇒ 'a ⇒ bool)" and tr: "transitiva R" and tot: "total R"
  shows "∃x. ∀y. y ≠ x ⟶ R x y"
proof -
  let ?r = "{(y, x). R y x}"
  have "wf ?r" using M5_finito_sin_regreso[OF fin irr tr] unfolding sin_regreso_def .
  then have "∃z∈(UNIV :: 'a set). ∀y. (y, z) ∈ ?r ⟶ y ∉ UNIV"
    by (rule wf_eq_minimal[THEN iffD1, rule_format, OF _ UNIV_I])
  then obtain x where "∀y. (y, x) ∈ ?r ⟶ y ∉ (UNIV :: 'a set)" by blast
  then have nadie: "∀y. ¬ R y x" by simp
  show ?thesis
  proof (intro exI allI impI)
    fix y assume "y ≠ x"
    then have "R y x ∨ R x y" using tot unfolding total_def by blast
    then show "R x y" using nadie by blast
  qed
qed

(* ---------- Lo que no se sigue (contraejemplos) ---------- *)

(* N1. Asimétrica no implica transitiva: 0 → 1 → 2, pero no 0 → 2. *)
lemma N1_asimetrica_no_transitiva:
  fixes R :: "nat ⇒ nat ⇒ bool"
  defines "R ≡ λx y. y = Suc x"
  shows "asimetrica R ∧ ¬ transitiva R"
proof
  show "asimetrica R" unfolding asimetrica_def R_def by simp
  show "¬ transitiva R"
  proof
    assume "transitiva R"
    then have "R 0 1 ∧ R 1 2 ⟶ R 0 2" unfolding transitiva_def by blast
    then show False unfolding R_def by simp
  qed
qed

(* N2. Irreflexiva no implica asimétrica: «es distinto de». *)
lemma N2_irreflexiva_no_asimetrica:
  fixes R :: "bool ⇒ bool ⇒ bool"
  defines "R ≡ λx y. x ≠ y"
  shows "irreflexiva R ∧ ¬ asimetrica R"
proof
  show "irreflexiva R" unfolding irreflexiva_def R_def by simp
  show "¬ asimetrica R"
  proof
    assume "asimetrica R"
    then have "R True False ⟶ ¬ R False True" unfolding asimetrica_def by blast
    then show False unfolding R_def by simp
  qed
qed

(* N3. Transitiva no implica irreflexiva: la relación que une todo con todo. *)
lemma N3_transitiva_no_irreflexiva:
  fixes R :: "bool ⇒ bool ⇒ bool"
  defines "R ≡ λx y. True"
  shows "transitiva R ∧ ¬ irreflexiva R"
  unfolding transitiva_def irreflexiva_def R_def by simp

(* N4. Con infinitas cosas, irreflexiva y transitiva no basta para que no haya regreso al infinito:
   en los números, a cada uno lo precede (en la relación) el siguiente, sin fin. Compárese con M5. *)
lemma N4_infinito_con_regreso:
  fixes R :: "nat ⇒ nat ⇒ bool"
  defines "R ≡ λx y. y < x"
  shows "irreflexiva R ∧ transitiva R ∧ ¬ sin_regreso R"
proof (intro conjI)
  show "irreflexiva R" unfolding irreflexiva_def R_def by simp
  show "transitiva R" unfolding transitiva_def R_def
  proof (intro allI impI)
    fix x y z :: nat assume "y < x ∧ z < y"
    then show "z < x" by (elim conjE) (rule less_trans)
  qed
  show "¬ sin_regreso R"
  proof
    assume "sin_regreso R"
    then have w: "wf {(y, x). R y x}" unfolding sin_regreso_def .
    have h: "∀i. (Suc i, i) ∈ {(y, x). R y x}" unfolding R_def by simp
    have "∃f. ∀i. (f (Suc i), f i) ∈ {(y, x). R y x}"
    proof (rule exI [of _ "λn. n"])
      show "∀i. ((λn. n) (Suc i), (λn. n) i) ∈ {(y, x). R y x}" using h by simp
    qed
    then show False using w unfolding wf_iff_no_infinite_down_chain by blast
  qed
qed

end
