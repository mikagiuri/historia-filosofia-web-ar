Theory Cinco_vias

theory Cinco_vias
  imports Main
begin

(* Cuarto mundo del Laboratorio (22-10-2026, propuesta del profesor): «Las cinco vías» (Tomás de Aquino,
   Suma de teología I, q. 2, a. 3), sobre el núcleo de los mundos anteriores. Debate abierto: se muestra qué
   premisa hace cada trabajo y dónde está cada objeción, sin veredicto sobre la existencia de Dios.
   Mismos nombres que la mesa web. *)

typedecl cosa

consts
  Mueve :: "cosa ⇒ cosa ⇒ bool"      (* x mueve a y: lo hace pasar de la potencia al acto (motus) *)
  Causa :: "cosa ⇒ cosa ⇒ bool"      (* x es causa eficiente de y *)
  ExisteEn :: "cosa ⇒ nat ⇒ bool"   (* tercera vía: x existe en el instante t *)
  MasPerfecto :: "cosa ⇒ cosa ⇒ bool"
  TiendeAFin :: "cosa ⇒ bool"
  Conoce :: "cosa ⇒ bool"
  Dirige :: "cosa ⇒ cosa ⇒ bool"

definition movido :: "cosa ⇒ bool" where "movido x ≡ ∃y. Mueve y x"
definition causado :: "cosa ⇒ bool" where "causado x ≡ ∃y. Causa y x"
definition contingente :: "cosa ⇒ bool" where "contingente x ≡ ∃t. ¬ ExisteEn x t"
definition necesario :: "cosa ⇒ bool" where "necesario x ≡ ∀t. ExisteEn x t"
definition maximo :: "cosa ⇒ bool" where "maximo x ≡ ∀y. y ≠ x ⟶ MasPerfecto x y"

(* ---------- Primera vía: el movimiento ---------- *)
definition V1_hay_movimiento :: bool where "V1_hay_movimiento ≡ ∃x. movido x"
definition V1_nada_se_mueve_a_si :: bool where "V1_nada_se_mueve_a_si ≡ ∀x. ¬ Mueve x x"
(* «No se puede proceder al infinito» en los motores POR SÍ (que mueven a la vez, como la mano el bastón):
   la relación «y mueve a x» está bien fundada, no hay cadenas descendentes infinitas. No es un principio del tiempo. *)
definition V1_sin_regreso :: bool where "V1_sin_regreso ≡ wf {(y, x). Mueve y x}"

(* Hay un primer motor que no es movido por nada. *)
theorem via1:
  assumes "V1_hay_movimiento" "V1_sin_regreso"
  shows "∃x. ¬ movido x ∧ (∃z. Mueve x z)"
proof -
  let ?R = "{(y, x). Mueve y x}"
  obtain m where "movido m" using assms(1) unfolding V1_hay_movimiento_def by blast
  then obtain y0 where "Mueve y0 m" unfolding movido_def by blast
  let ?S = "{x. ∃z. Mueve x z}"
  have "y0 ∈ ?S" using ‹Mueve y0 m› by blast
  have "∃z∈?S. ∀y. (y, z) ∈ ?R ⟶ y ∉ ?S"
    using wf_eq_minimal[THEN iffD1, OF assms(2)[unfolded V1_sin_regreso_def], rule_format, OF ‹y0 ∈ ?S›] .
  then obtain x where "x ∈ ?S" and min: "∀y. (y, x) ∈ ?R ⟶ y ∉ ?S" by blast
  have "¬ movido x" using min unfolding movido_def by blast
  then show ?thesis using ‹x ∈ ?S› by blast
qed

(* «Nada se mueve a sí mismo» no hace falta suponerlo: sale de que no hay regreso al infinito. *)
theorem via1_irreflexiva:
  assumes "V1_sin_regreso"
  shows "¬ Mueve x x"
proof
  assume "Mueve x x"
  have w: "wf {(y, x). Mueve y x}" using assms unfolding V1_sin_regreso_def .
  have "(x, x) ∉ {(y, x). Mueve y x}" using w by (rule wf_not_refl)
  then show False using ‹Mueve x x› by simp
qed

(* Sin la prohibición del regreso al infinito, no se sigue: cada número lo mueve el siguiente. *)
lemma via1_sin_regreso_contramodelo:
  fixes M :: "nat ⇒ nat ⇒ bool"
  defines "M ≡ λy x. y = Suc x"
  shows "(∃x. ∃y. M y x) ∧ (∀x. ¬ M x x) ∧ ¬ (∃x. ¬ (∃y. M y x) ∧ (∃z. M x z))"
  unfolding M_def by simp

(* En un mundo finito, el regreso al infinito queda excluido solo si no hay círculos de motores. *)
lemma finito_sin_circulos_es_sin_regreso:
  assumes "finite (UNIV :: cosa set)" "acyclic {(y, x). Mueve y x}"
  shows "V1_sin_regreso"
proof -
  have "finite (UNIV :: (cosa × cosa) set)" using assms(1) by (simp add: finite_Prod_UNIV)
  then have "finite {(y, x). Mueve y x}" by (rule finite_subset[rotated]) simp
  then show ?thesis unfolding V1_sin_regreso_def using assms(2) by (simp add: finite_acyclic_wf)
qed

(* ---------- Segunda vía: la causa eficiente (la misma estructura) ---------- *)
definition V2_hay_causas :: bool where "V2_hay_causas ≡ ∃x. causado x"
definition V2_sin_regreso :: bool where "V2_sin_regreso ≡ wf {(y, x). Causa y x}"

theorem via2:
  assumes "V2_hay_causas" "V2_sin_regreso"
  shows "∃x. ¬ causado x ∧ (∃z. Causa x z)"
proof -
  obtain m y0 where "Causa y0 m" using assms(1) unfolding V2_hay_causas_def causado_def by blast
  let ?S = "{x. ∃z. Causa x z}"
  have "y0 ∈ ?S" using ‹Causa y0 m› by blast
  have "∃z∈?S. ∀y. (y, z) ∈ {(y, x). Causa y x} ⟶ y ∉ ?S"
    using wf_eq_minimal[THEN iffD1, OF assms(2)[unfolded V2_sin_regreso_def], rule_format, OF ‹y0 ∈ ?S›] .
  then obtain x where "x ∈ ?S" and min: "∀y. (y, x) ∈ {(y, x). Causa y x} ⟶ y ∉ ?S" by blast
  then show ?thesis unfolding causado_def by blast
qed

(* ---------- Tercera vía: lo contingente y lo necesario ---------- *)
definition V3_siempre_algo :: bool where "V3_siempre_algo ≡ ∀t. ∃x. ExisteEn x t"
(* El paso discutido: «si todo puede no ser, en algún momento no fue nada». *)
definition V3_paso_de_Tomas :: bool where "V3_paso_de_Tomas ≡ (∀x. contingente x) ⟶ (∃t. ∀x. ¬ ExisteEn x t)"

(* Con el paso de Tomás, si siempre hay algo, hay algo necesario. *)
theorem via3:
  assumes "V3_siempre_algo" "V3_paso_de_Tomas"
  shows "∃x. necesario x"
proof (rule ccontr)
  assume "¬ (∃x. necesario x)"
  then have "∀x. contingente x" unfolding necesario_def contingente_def by simp
  then obtain t where vacio: "∀x. ¬ ExisteEn x t" using assms(2) unfolding V3_paso_de_Tomas_def by blast
  obtain x where "ExisteEn x t" using assms(1) unfolding V3_siempre_algo_def by blast
  then show False using vacio by simp
qed

(* Pero el paso es un salto de cuantificadores: que cada cosa deje de existir en algún instante no obliga a que
   haya un instante en que no existe nada. Dos cosas que se relevan: la primera existe solo en 0, la segunda desde 1. *)
lemma via3_salto:
  fixes E :: "bool ⇒ nat ⇒ bool"
  defines "E ≡ λx t. if x then t = 0 else t ≥ 1"
  shows "(∀x. ∃t. ¬ E x t) ∧ (∀t. ∃x. E x t) ∧ ¬ (∃x. ∀t. E x t)"
proof (intro conjI allI)
  fix x :: bool
  show "∃t. ¬ E x t"
  proof (cases x)
    case True then show ?thesis unfolding E_def by (intro exI [of _ 1]) simp
  next
    case False then show ?thesis unfolding E_def by (intro exI [of _ 0]) simp
  qed
next
  fix t :: nat
  show "∃x. E x t"
  proof (cases "t = 0")
    case True then show ?thesis unfolding E_def by (intro exI [of _ True]) simp
  next
    case False then show ?thesis unfolding E_def by (intro exI [of _ False]) simp
  qed
next
  show "¬ (∃x. ∀t. E x t)"
  proof
    assume "∃x. ∀t. E x t"
    then obtain x where h: "∀t. E x t" by blast
    then have "E x 0" and "E x 1" by blast+
    then show False unfolding E_def by (cases x) simp_all
  qed
qed

(* ---------- Cuarta vía: los grados ---------- *)
definition V4_orden :: bool where "V4_orden ≡ (∀x. ¬ MasPerfecto x x) ∧ (∀x y z. MasPerfecto x y ∧ MasPerfecto y z ⟶ MasPerfecto x z)"
definition V4_comparables :: bool where "V4_comparables ≡ ∀x y. x ≠ y ⟶ MasPerfecto x y ∨ MasPerfecto y x"

(* Si el orden es estricto y todo es comparable, hay como mucho un máximo. *)
theorem via4_unico:
  assumes "V4_orden"
  shows "maximo x ∧ maximo y ⟶ x = y"
proof (intro impI, elim conjE, rule ccontr)
  assume mx: "maximo x" and my: "maximo y" and d: "x ≠ y"
  have "MasPerfecto x y" using mx d unfolding maximo_def by simp
  moreover have "MasPerfecto y x" using my d unfolding maximo_def by simp
  moreover have tr: "∀a b c. MasPerfecto a b ∧ MasPerfecto b c ⟶ MasPerfecto a c" using assms unfolding V4_orden_def by (rule conjunct2)
  ultimately have "MasPerfecto x x" by blast
  then show False using assms unfolding V4_orden_def by simp
qed

(* En un mundo finito, si todo es comparable, hay un máximo. *)
theorem via4_finito:
  assumes fin: "finite (UNIV :: cosa set)" and "V4_orden" "V4_comparables"
  shows "∃x. maximo x"
proof -
  let ?R = "{(y, x). MasPerfecto y x}"
  have irr: "∀x. ¬ MasPerfecto x x" and tr: "∀x y z. MasPerfecto x y ∧ MasPerfecto y z ⟶ MasPerfecto x z"
    using assms(2) unfolding V4_orden_def by blast+
  have "acyclic ?R"
  proof -
    have cl: "(a, b) ∈ ?R⇧+ ⟹ MasPerfecto a b" for a b
    proof (induction rule: trancl_induct)
      case (base y) then show ?case by simp
    next
      case (step y z) then show ?case using tr by blast
    qed
    show ?thesis unfolding acyclic_def using cl irr by blast
  qed
  moreover have "finite ?R"
  proof -
    have "finite (UNIV :: (cosa × cosa) set)" using fin by (simp add: finite_Prod_UNIV)
    then show ?thesis by (rule finite_subset[rotated]) simp
  qed
  ultimately have "wf ?R" by (simp add: finite_acyclic_wf)
  then have "∃z∈(UNIV :: cosa set). ∀y. (y, z) ∈ ?R ⟶ y ∉ UNIV"
    by (rule wf_eq_minimal[THEN iffD1, rule_format, OF _ UNIV_I])
  then obtain x where "∀y. (y, x) ∈ ?R ⟶ y ∉ (UNIV :: cosa set)" by blast
  then have "∀y. ¬ MasPerfecto y x" by simp
  then show ?thesis using assms(3) unfolding V4_comparables_def maximo_def by blast
qed

(* Que exista un máximo depende de que el mundo sea finito (o de otra premisa): con los naturales, siempre hay uno más. *)
lemma via4_sin_maximo:
  fixes P :: "nat ⇒ nat ⇒ bool"
  defines "P ≡ λx y. y < x"
  shows "(∀x. ¬ P x x) ∧ (∀x y z. P x y ∧ P y z ⟶ P x z) ∧ (∀x y. x ≠ y ⟶ P x y ∨ P y x)
         ∧ ¬ (∃x. ∀y. y ≠ x ⟶ P x y)"
  unfolding P_def by (auto, metis Suc_n_not_le_n le_eq_less_or_eq lessI)

(* ---------- Quinta vía: el gobierno del mundo ---------- *)
definition V5_dirigido :: bool where "V5_dirigido ≡ ∀x. TiendeAFin x ∧ ¬ Conoce x ⟶ (∃y. Conoce y ∧ Dirige y x)"
definition V5_hay_fines :: bool where "V5_hay_fines ≡ ∃x. TiendeAFin x ∧ ¬ Conoce x"

(* Toda la fuerza está en la premisa: con ella, hay alguien que conoce y dirige. *)
theorem via5:
  assumes "V5_hay_fines" "V5_dirigido"
  shows "∃y. Conoce y ∧ (∃x. Dirige y x)"
  using assms unfolding V5_hay_fines_def V5_dirigido_def by blast

(* ---------- «Y a esto llamamos Dios»: ¿es lo mismo en las cinco? ---------- *)
(* Que el primer motor y la primera causa sean la misma cosa no se sigue: puede haber un primer motor y otra primera causa. *)
lemma identidad_no_se_sigue:
  "V1_hay_movimiento ⟹ V1_nada_se_mueve_a_si ⟹ V2_hay_causas ⟹
   ¬ movido x ⟹ Mueve x z ⟹ ¬ causado y ⟹ Causa y w ⟹ x = y"
  nitpick [expect = genuine, timeout = 120]
  oops

end
