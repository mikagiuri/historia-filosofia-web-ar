Theory Mundo_tiempo_creacion

theory Mundo_tiempo_creacion
  imports Main
begin

(* Mundo de práctica del Laboratorio (21-10-2026): cosas que se distinguen por la precedencia
   en el tiempo, la duración y si unas fueron creadas por otras. Mismos nombres que la mesa web. *)

typedecl cosa

consts
  Antes :: "cosa ⇒ cosa ⇒ bool"
  DuraMas :: "cosa ⇒ cosa ⇒ bool"
  Crea :: "cosa ⇒ cosa ⇒ bool"

definition T_irreflexiva :: bool where "T_irreflexiva ≡ ∀x. ¬ Antes x x"
definition T_transitiva :: bool where "T_transitiva ≡ ∀x y z. Antes x y ∧ Antes y z ⟶ Antes x z"
definition D_irreflexiva :: bool where "D_irreflexiva ≡ ∀x. ¬ DuraMas x x"
definition D_transitiva :: bool where "D_transitiva ≡ ∀x y z. DuraMas x y ∧ DuraMas y z ⟶ DuraMas x z"
definition C_creador_anterior :: bool where "C_creador_anterior ≡ ∀x y. Crea x y ⟶ Antes x y"
definition C_un_creador :: bool where "C_un_creador ≡ ∀x y z. Crea x z ∧ Crea y z ⟶ x = y"
definition C_creador_dura_mas :: bool where "C_creador_dura_mas ≡ ∀x y. Crea x y ⟶ DuraMas x y"
definition C_todo_creado :: bool where "C_todo_creado ≡ ∀x. ∃y. Crea y x"

definition increado :: "cosa ⇒ bool" where "increado x ≡ ¬ (∃y. Crea y x)"

(* Q1: Nada se crea a sí mismo. *)
theorem Q1:
  assumes "C_creador_anterior" "T_irreflexiva"
  shows "¬ Crea x x"
  using assms unfolding C_creador_anterior_def T_irreflexiva_def by blast

(* Q1 por otro camino, el de la duración (propuesto por el profesor, 21-10): si x se creara a sí mismo,
   duraría más que sí mismo. Usa la premisa discutible: si se quita, este camino se cae y queda el del tiempo. *)
theorem Q1_por_duracion:
  assumes "C_creador_dura_mas" "D_irreflexiva"
  shows "¬ Crea x x"
  using assms unfolding C_creador_dura_mas_def D_irreflexiva_def by blast

lemma Q1_sin_irreflexiva: "C_creador_anterior ⟹ ¬ Crea x x"
  nitpick [expect = genuine, timeout = 120]
  oops

(* Q2: Si x crea a y, y no crea a x. *)
theorem Q2:
  assumes "C_creador_anterior" "T_irreflexiva" "T_transitiva"
  shows "Crea x y ⟶ ¬ Crea y x"
  using assms unfolding C_creador_anterior_def T_irreflexiva_def T_transitiva_def by blast

lemma Q2_sin_transitiva: "C_creador_anterior ⟹ T_irreflexiva ⟹ Crea x y ⟶ ¬ Crea y x"
  nitpick [expect = genuine, timeout = 120]
  oops

(* Q3: El creador del creador es anterior a lo creado. *)
theorem Q3:
  assumes "C_creador_anterior" "T_transitiva"
  shows "Crea x y ∧ Crea y z ⟶ Antes x z"
  using assms unfolding C_creador_anterior_def T_transitiva_def by blast

(* Q4: Lo creado no dura más que su creador. *)
theorem Q4:
  assumes "C_creador_dura_mas" "D_irreflexiva" "D_transitiva"
  shows "Crea x y ⟶ ¬ DuraMas y x"
  using assms unfolding C_creador_dura_mas_def D_irreflexiva_def D_transitiva_def by blast

lemma Q4_sin_dura_mas:
  "T_irreflexiva ⟹ T_transitiva ⟹ D_irreflexiva ⟹ D_transitiva ⟹ C_creador_anterior ⟹ C_un_creador ⟹
   Crea x y ⟶ ¬ DuraMas y x"
  nitpick [expect = genuine, timeout = 120]
  oops

(* Q8 (en la mesa, la 5): Lo creado puede sobrevivir a su creador. Es una POSIBILIDAD: hay un mundo
   que cumple todas las premisas salvo la discutible (y «todo ha sido creado») en el que algo creado
   dura más que quien lo creó: el autor que muere y el libro que queda. Mundo de dos cosas:
   True (el autor) crea y empieza antes que False (el libro), y False dura más que True. *)
lemma Q8_posible:
  fixes A D C :: "bool ⇒ bool ⇒ bool"
  defines "A ≡ λx y. x ∧ ¬ y" and "D ≡ λx y. ¬ x ∧ y" and "C ≡ λx y. x ∧ ¬ y"
  shows "(∀x. ¬ A x x) ∧ (∀x y z. A x y ∧ A y z ⟶ A x z)
         ∧ (∀x. ¬ D x x) ∧ (∀x y z. D x y ∧ D y z ⟶ D x z)
         ∧ (∀x y. C x y ⟶ A x y) ∧ (∀x y z. C x z ∧ C y z ⟶ x = y)
         ∧ (∃x y. C x y ∧ D y x)"
  unfolding A_def D_def C_def by auto

(* Y con la premisa discutible ya no es posible: es Q4. *)
corollary Q8_imposible_con_dura_mas:
  assumes "C_creador_dura_mas" "D_irreflexiva" "D_transitiva"
  shows "¬ (∃x y. Crea x y ∧ DuraMas y x)"
  using Q4 [OF assms] by blast

(* Q9 (en la mesa, la 9): Algo puede tener dos creadores. Otra POSIBILIDAD: sin «nada tiene dos
   creadores» hay un mundo, con todas las demás premisas compatibles (también la discutible), en el que
   dos cosas distintas crean a una tercera. Hacen falta tres cosas: 0 y 1 crean a 2. *)
lemma Q9_posible:
  fixes A D C :: "nat ⇒ nat ⇒ bool"
  defines "A ≡ λx y. x < y" and "C ≡ λx y. x < 2 ∧ y = 2" and "D ≡ λx y. x < 2 ∧ y = 2"
  shows "(∀x. ¬ A x x) ∧ (∀x y z. A x y ∧ A y z ⟶ A x z)
         ∧ (∀x. ¬ D x x) ∧ (∀x y z. D x y ∧ D y z ⟶ D x z)
         ∧ (∀x y. C x y ⟶ A x y) ∧ (∀x y. C x y ⟶ D x y)
         ∧ (∃x y z. x ≠ y ∧ C x z ∧ C y z)"
  unfolding A_def C_def D_def
  apply (intro conjI)
        apply auto[6]
  by (rule exI [of _ 0], rule exI [of _ 1], rule exI [of _ 2]) simp

(* Con «nada tiene dos creadores» no puede ocurrir: es su definición. *)
lemma Q9_imposible_con_un_creador:
  assumes "C_un_creador"
  shows "¬ (∃x y z. x ≠ y ∧ Crea x z ∧ Crea y z)"
  using assms unfolding C_un_creador_def by blast

(* Q5: Lo anterior dura más. No se sigue ni con todas las premisas (salvo la incompatible). *)
lemma Q5_no_se_sigue:
  "T_irreflexiva ⟹ T_transitiva ⟹ D_irreflexiva ⟹ D_transitiva ⟹ C_creador_anterior ⟹ C_un_creador ⟹
   C_creador_dura_mas ⟹ Antes x y ⟶ DuraMas x y"
  nitpick [expect = genuine, timeout = 120]
  oops

(* Q6: Hay algo increado. Vale en todo mundo FINITO... *)
theorem Q6_finito:
  assumes fin: "finite (UNIV :: cosa set)"
    and "C_creador_anterior" "T_irreflexiva" "T_transitiva"
  shows "∃x. increado x"
proof -
  let ?R = "{(y, x). Crea y x}"
  have ant: "⋀a b. (a, b) ∈ ?R⇧+ ⟹ Antes a b"
  proof -
    fix a b assume "(a, b) ∈ ?R⇧+"
    then show "Antes a b"
    proof (induction rule: trancl_induct)
      case (base y) then show ?case using assms(2) unfolding C_creador_anterior_def by blast
    next
      case (step y z) then show ?case using assms(2,4) unfolding C_creador_anterior_def T_transitiva_def by blast
    qed
  qed
  have "acyclic ?R" unfolding acyclic_def using ant assms(3) unfolding T_irreflexiva_def by blast
  moreover have "finite ?R"
  proof -
    have "finite (UNIV :: (cosa × cosa) set)" using fin by (simp add: finite_Prod_UNIV)
    then show ?thesis by (rule finite_subset[rotated]) simp
  qed
  ultimately have "wf ?R" by (simp add: finite_acyclic_wf)
  then obtain x where "x ∈ (UNIV :: cosa set)" "∀y. (y, x) ∈ ?R ⟶ y ∉ UNIV"
    using wfE_min [of ?R "undefined" UNIV] by blast
  then show ?thesis unfolding increado_def by blast
qed

(* ... y por eso, en un mundo finito, «todo ha sido creado» es incompatible con las tres premisas. *)
corollary todo_creado_incompatible_finito:
  assumes "finite (UNIV :: cosa set)" "C_creador_anterior" "T_irreflexiva" "T_transitiva"
  shows "¬ C_todo_creado"
  using Q6_finito [OF assms] unfolding increado_def C_todo_creado_def by blast

(* ... pero no en uno infinito: los números naturales, donde n + 1 crea a n y lo mayor es anterior.
   Cumple las premisas de tiempo y creación, y todo ha sido creado: no hay nada increado. *)
lemma Q6_contramodelo_infinito:
  fixes A C :: "nat ⇒ nat ⇒ bool"
  defines "A ≡ λx y. y < x" and "C ≡ λx y. x = Suc y"
  shows "(∀x. ¬ A x x) ∧ (∀x y z. A x y ∧ A y z ⟶ A x z) ∧ (∀x y. C x y ⟶ A x y)
         ∧ (∀x y z. C x z ∧ C y z ⟶ x = y) ∧ (∀x. ∃y. C y x) ∧ ¬ (∃x. ¬ (∃y. C y x))"
  unfolding A_def C_def by auto

(* Q7: Hay un único increado. No se sigue: puede haber dos orígenes. *)
lemma Q7_no_se_sigue:
  "T_irreflexiva ⟹ T_transitiva ⟹ D_irreflexiva ⟹ D_transitiva ⟹ C_creador_anterior ⟹ C_un_creador ⟹
   C_creador_dura_mas ⟹ increado x ⟹ increado y ⟹ x = y"
  nitpick [expect = genuine, timeout = 120]
  oops

end
