section ‹Tercera antinomia: libertad y causalidad natural (A444-451 / B472-479)›

typedecl suceso
consts Causa :: "suceso ⇒ suceso ⇒ bool"   (* Causa y x: el estado y determina a x según una ley natural *)

(* Libre, en sentido cosmológico: comienza por sí una serie; ningún estado anterior lo determina. *)
definition libre :: "suceso ⇒ bool" where "libre x ≡ ¬ (∃y. Causa y x)"

(* Premisa propia de la tesis: la serie de las causas de lo que sucede está completa (no hay regreso al infinito). *)
definition T3_serie_completa :: bool where "T3_serie_completa ≡ wf {(y, x). Causa y x}"
(* Premisa propia de la antítesis: la ley natural, todo lo que sucede tiene un estado anterior que lo determina. *)
definition A3_ley_natural :: bool where "A3_ley_natural ≡ ∀x. ∃y. Causa y x"

(* Tesis: si la serie está completa, hay causalidad por libertad (algún suceso no lo determina ningún estado anterior). *)
theorem tesis3:
  assumes "T3_serie_completa"
  shows "∃x. libre x"
proof -
  let ?r = "{(y, x). Causa y x}"
  have w: "wf ?r" using assms unfolding T3_serie_completa_def .
  then have "∃z∈(UNIV :: suceso set). ∀y. (y, z) ∈ ?r ⟶ y ∉ UNIV"
    by (rule wf_eq_minimal[THEN iffD1, rule_format, OF _ UNIV_I])
  then obtain x where "∀y. (y, x) ∈ ?r ⟶ y ∉ (UNIV :: suceso set)" by blast
  then have "¬ (∃y. Causa y x)" by simp
  then show ?thesis unfolding libre_def by blast
qed

(* Antítesis: si rige la ley natural, no hay libertad. *)
theorem antitesis3:
  assumes "A3_ley_natural"
  shows "¬ (∃x. libre x)"
  using assms unfolding A3_ley_natural_def libre_def by blast

(* Las dos premisas, sobre el mismo dominio de sucesos, no caben juntas. *)
theorem choque3:
  assumes "T3_serie_completa" "A3_ley_natural"
  shows False
  using tesis3[OF assms(1)] antitesis3[OF assms(2)] by blast

(* Cada lado, por separado, tiene un modelo: la tesis, con los naturales (el 0 no tiene causa); la antítesis, con los
   enteros (cada suceso tiene una causa anterior, sin fin hacia atrás). *)
lemma modelo_tesis3:
  fixes C :: "nat ⇒ nat ⇒ bool"
  defines "C ≡ λy x. x = Suc y"
  shows "wf {(y, x). C y x} ∧ ¬ (∀x. ∃y. C y x)"
proof (rule conjI)
  have "{(y, x). C y x} = pred_nat" unfolding C_def pred_nat_def by simp
  then show "wf {(y, x). C y x}" using wf_pred_nat by simp
  show "¬ (∀x. ∃y. C y x)"
  proof
    assume "∀x. ∃y. C y x"
    then obtain y where "C y 0" by blast
    then show False unfolding C_def by simp
  qed
qed

lemma modelo_antitesis3:
  fixes C :: "int ⇒ int ⇒ bool"
  defines "C ≡ λy x. x = y + 1"
  shows "(∀x. ∃y. C y x) ∧ ¬ wf {(y, x). C y x}"
proof (rule conjI)
  show "∀x. ∃y. C y x"
  proof
    fix x :: int
    show "∃y. C y x" unfolding C_def by (rule exI [of _ "x - 1"]) simp
  qed
  show "¬ wf {(y, x). C y x}"
  proof
    assume w: "wf {(y, x). C y x}"
    have h: "∀i. (- int (Suc i), - int i) ∈ {(y, x). C y x}" unfolding C_def by simp
    have "∃f. ∀i. (f (Suc i), f i) ∈ {(y, x). C y x}"
    proof (rule exI [of _ "λn. - int n"])
      show "∀i. ((λn. - int n) (Suc i), (λn. - int n) i) ∈ {(y, x). C y x}" using h by simp
    qed
    then show False using w unfolding wf_iff_no_infinite_down_chain by blast
  qed
qed

(* La solución de Kant (A532-558 / B560-586): un mismo efecto puede tener una causa empírica, entre fenómenos, y una
   causa inteligible, que no es un fenómeno. Modelo: los fenómenos son Some k (k entero) y lo inteligible es None.
   Causa natural (C): de Some k a Some (k+1). Causa inteligible (I): de None a cada fenómeno.
   Se cumplen a la vez la antítesis para los fenómenos (todo fenómeno tiene causa natural entre fenómenos) y la tesis
   para lo inteligible (hay algo sin causa natural que es causa, inteligible, de los fenómenos). *)
lemma solucion3:
  fixes F :: "int option ⇒ bool" and C I :: "int option ⇒ int option ⇒ bool"
  defines "F ≡ λx. x ≠ None"
    and "C ≡ λy x. ∃k. y = Some k ∧ x = Some (k + 1)"
    and "I ≡ λy x. y = None ∧ x ≠ None"
  shows "(∀x. F x ⟶ (∃y. F y ∧ C y x)) ∧ (∃x. ¬ F x ∧ ¬ (∃y. C y x) ∧ (∀z. F z ⟶ I x z))"
proof (rule conjI)
  show "∀x. F x ⟶ (∃y. F y ∧ C y x)"
  proof (intro allI impI)
    fix x :: "int option"
    assume "F x"
    then obtain k where k: "x = Some k" unfolding F_def by (cases x) simp_all
    show "∃y. F y ∧ C y x"
    proof (rule exI [of _ "Some (k - 1)"])
      show "F (Some (k - 1)) ∧ C (Some (k - 1)) x" unfolding F_def C_def k by simp
    qed
  qed
  show "∃x. ¬ F x ∧ ¬ (∃y. C y x) ∧ (∀z. F z ⟶ I x z)"
  proof (rule exI [of _ None])
    show "¬ F None ∧ ¬ (∃y. C y None) ∧ (∀z. F z ⟶ I None z)" unfolding F_def C_def I_def by simp
  qed
qed

section ‹Cuarta antinomia: el ser necesario (A452-461 / B480-489)›

typedecl ser
consts Cond :: "ser ⇒ ser ⇒ bool"   (* Cond y x: y es condición de la existencia de x *)
consts EnMundo :: "ser ⇒ bool"          (* x pertenece al mundo sensible: está en el tiempo *)

(* Absolutamente necesario: incondicionado, su existencia no depende de ninguna condición. *)
definition incondicionado :: "ser ⇒ bool" where "incondicionado x ≡ ¬ (∃y. Cond y x)"

(* Premisa propia de la tesis: la serie de las condiciones está completa (no hay regreso al infinito). *)
definition T4_serie_completa :: bool where "T4_serie_completa ≡ wf {(y, x). Cond y x}"
(* Premisa que usan las dos partes: lo que condiciona algo que está en el tiempo está también en el tiempo. *)
definition T4_condicion_en_el_tiempo :: bool where
  "T4_condicion_en_el_tiempo ≡ ∀x z. Cond x z ∧ EnMundo z ⟶ EnMundo x"
(* Premisa propia de la antítesis: la ley dinámica, todo lo que hay en el mundo tiene una condición. *)
definition A4_ley_dinamica :: bool where "A4_ley_dinamica ≡ ∀x. EnMundo x ⟶ (∃y. Cond y x)"

(* Tesis: con la serie completa, si hay mundo, al mundo pertenece algo absolutamente necesario. *)
theorem tesis4:
  assumes "T4_serie_completa" "T4_condicion_en_el_tiempo" "∃z. EnMundo z"
  shows "∃x. EnMundo x ∧ incondicionado x"
proof -
  let ?r = "{(y, x). Cond y x}" and ?S = "{x. EnMundo x}"
  obtain z where "EnMundo z" using assms(3) by blast
  then have "z ∈ ?S" by simp
  have "∃m∈?S. ∀y. (y, m) ∈ ?r ⟶ y ∉ ?S"
    using wf_eq_minimal[THEN iffD1, OF assms(1)[unfolded T4_serie_completa_def], rule_format, OF ‹z ∈ ?S›] .
  then obtain m where m: "m ∈ ?S" and min: "∀y. (y, m) ∈ ?r ⟶ y ∉ ?S" by blast
  have "incondicionado m" unfolding incondicionado_def
  proof
    assume "∃y. Cond y m"
    then obtain y where c: "Cond y m" by blast
    have "EnMundo m" using m by simp
    then have "EnMundo y" using c assms(2) unfolding T4_condicion_en_el_tiempo_def by blast
    moreover have "(y, m) ∈ ?r" using c by simp
    ultimately show False using min by simp
  qed
  moreover have "EnMundo m" using m by simp
  ultimately show ?thesis by blast
qed

(* Antítesis, primera parte: con la ley dinámica, nada en el mundo es incondicionado. *)
theorem antitesis4_en_el_mundo:
  assumes "A4_ley_dinamica"
  shows "¬ (∃x. EnMundo x ∧ incondicionado x)"
  using assms unfolding A4_ley_dinamica_def incondicionado_def by blast

(* Antítesis, segunda parte: fuera del mundo tampoco, porque lo que condiciona algo del mundo estaría en el tiempo. *)
theorem antitesis4_fuera:
  assumes "T4_condicion_en_el_tiempo"
  shows "¬ (∃x. ¬ EnMundo x ∧ (∃z. EnMundo z ∧ Cond x z))"
  using assms unfolding T4_condicion_en_el_tiempo_def by blast

theorem choque4:
  assumes "T4_serie_completa" "T4_condicion_en_el_tiempo" "A4_ley_dinamica" "∃z. EnMundo z"
  shows False
  using tesis4[OF assms(1) assms(2) assms(4)] antitesis4_en_el_mundo[OF assms(3)] by blast

(* El paso del argumento cosmológico, que Kant examina aquí: de «cada miembro de la serie está condicionado» no se
   sigue «la serie entera tiene una condición fuera de ella». Una serie (un conjunto S) está condicionada como un todo si
   algo de fuera de S condiciona algo de S. La serie de todo lo que hay no lo está nunca, aunque cada miembro tenga su
   condición: hace falta una premisa más. Es lógica de orden superior: se habla de conjuntos de seres. *)
definition serie_condicionada :: "ser set ⇒ bool" where
  "serie_condicionada S ≡ ∃y. y ∉ S ∧ (∃x∈S. Cond y x)"

lemma el_todo_no_tiene_condicion_exterior: "¬ serie_condicionada UNIV"
  unfolding serie_condicionada_def by simp

lemma cada_miembro_no_es_el_todo:
  assumes "∀x. ∃y. Cond y x"
  shows "(∀x∈(UNIV :: ser set). ∃y. Cond y x) ∧ ¬ serie_condicionada UNIV"
  using assms el_todo_no_tiene_condicion_exterior by blast

(* Con la serie completa, la otra salida (nada necesario ni en el mundo ni fuera de él) es imposible, aunque no se
   use la premisa «la condición está en el tiempo»: basta con que nada de fuera condicione el mundo. *)
lemma salida4_con_serie_completa:
  assumes "T4_serie_completa" "∃z. EnMundo z" "∀x. ¬ EnMundo x ⟶ ¬ (∃y. EnMundo y ∧ Cond x y)"
  shows "∃x. EnMundo x ∧ incondicionado x"
proof -
  let ?r = "{(y, x). Cond y x}" and ?S = "{x. EnMundo x}"
  obtain z where "EnMundo z" using assms(2) by blast
  then have "z ∈ ?S" by simp
  have "∃m∈?S. ∀y. (y, m) ∈ ?r ⟶ y ∉ ?S"
    using wf_eq_minimal[THEN iffD1, OF assms(1)[unfolded T4_serie_completa_def], rule_format, OF ‹z ∈ ?S›] .
  then obtain m where m: "m ∈ ?S" and min: "∀y. (y, m) ∈ ?r ⟶ y ∉ ?S" by blast
  have "incondicionado m" unfolding incondicionado_def
  proof
    assume "∃y. Cond y m"
    then obtain y where c: "Cond y m" by blast
    have "EnMundo m" using m by simp
    then have "EnMundo y" using c assms(3) by blast
    moreover have "(y, m) ∈ ?r" using c by simp
    ultimately show False using min by simp
  qed
  moreover have "EnMundo m" using m by simp
  ultimately show ?thesis by blast
qed

(* Otra salida, no la de Kant: soltar la premisa de la tesis (la serie completa). Con un regreso infinito de
   condiciones, todo en el mundo está condicionado y nada es necesario, sin contradicción: es la objeción de Cleantes en los
   Diálogos de Hume (IX), y la que se sigue del infinito actual de Cantor. Modelo: los enteros, cada uno condicionado por
   el anterior, todos en el mundo. *)
lemma modelo_antitesis4:
  fixes M :: "int ⇒ bool" and C :: "int ⇒ int ⇒ bool"
  defines "M ≡ λx. True"
    and "C ≡ λy x. x = y + 1"
  shows "(∀x z. C x z ∧ M z ⟶ M x) ∧ (∀x. M x ⟶ (∃y. C y x)) ∧ (∃z. M z)
         ∧ ¬ (∃x. M x ∧ ¬ (∃y. C y x)) ∧ ¬ wf {(y, x). C y x}"
proof (intro conjI)
  show "∀x z. C x z ∧ M z ⟶ M x" unfolding M_def by simp
  show "∀x. M x ⟶ (∃y. C y x)"
  proof (intro allI impI)
    fix x :: int
    show "∃y. C y x" unfolding C_def by (rule exI [of _ "x - 1"]) simp
  qed
  show "∃z. M z" unfolding M_def by simp
  show "¬ (∃x. M x ∧ ¬ (∃y. C y x))"
  proof
    assume "∃x. M x ∧ ¬ (∃y. C y x)"
    then obtain x where n: "¬ (∃y. C y x)" by blast
    have "C (x - 1) x" unfolding C_def by simp
    then show False using n by blast
  qed
  show "¬ wf {(y, x). C y x}"
  proof
    assume w: "wf {(y, x). C y x}"
    have h: "∀i. (- int (Suc i), - int i) ∈ {(y, x). C y x}" unfolding C_def by simp
    have "∃f. ∀i. (f (Suc i), f i) ∈ {(y, x). C y x}"
    proof (rule exI [of _ "λn. - int n"])
      show "∀i. ((λn. - int n) (Suc i), (λn. - int n) i) ∈ {(y, x). C y x}" using h by simp
    qed
    then show False using w unfolding wf_iff_no_infinite_down_chain by blast
  qed
qed

(* La solución de Kant: el ser necesario no es un miembro de la serie empírica sino su fundamento inteligible, fuera
   del tiempo. Modelo: en el mundo, Some k; fuera, None. Condición empírica (E): de Some k a Some (k+1); fundamento
   inteligible (G): de None a todo lo del mundo. Para E vale «la condición de algo del mundo está en el mundo» y la ley
   dinámica; G no es una condición en el tiempo. *)
lemma solucion4:
  fixes M :: "int option ⇒ bool" and E G :: "int option ⇒ int option ⇒ bool"
  defines "M ≡ λx. x ≠ None"
    and "E ≡ λy x. ∃k. y = Some k ∧ x = Some (k + 1)"
    and "G ≡ λy x. y = None ∧ x ≠ None"
  shows "(∀x z. E x z ∧ M z ⟶ M x) ∧ (∀x. M x ⟶ (∃y. E y x))
         ∧ (∃n. ¬ M n ∧ ¬ (∃y. E y n) ∧ (∀z. M z ⟶ G n z))"
proof (intro conjI)
  show "∀x z. E x z ∧ M z ⟶ M x"
  proof (intro allI impI)
    fix x z :: "int option"
    assume "E x z ∧ M z"
    then obtain k where "x = Some k" unfolding E_def by blast
    then show "M x" unfolding M_def by simp
  qed
  show "∀x. M x ⟶ (∃y. E y x)"
  proof (intro allI impI)
    fix x :: "int option"
    assume "M x"
    then obtain k where k: "x = Some k" unfolding M_def by (cases x) simp_all
    show "∃y. E y x"
    proof (rule exI [of _ "Some (k - 1)"])
      show "E (Some (k - 1)) x" unfolding E_def k by simp
    qed
  qed
  show "∃n. ¬ M n ∧ ¬ (∃y. E y n) ∧ (∀z. M z ⟶ G n z)"
  proof (rule exI [of _ None])
    show "¬ M None ∧ ¬ (∃y. E y None) ∧ (∀z. M z ⟶ G None z)" unfolding M_def E_def G_def by simp
  qed
qed
