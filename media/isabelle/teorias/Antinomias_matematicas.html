(* Quinto mundo del Laboratorio (22-10-2026, propuesta del profesor): las antinomias de Kant. Aquí las matemáticas;
   las dinámicas están en Antinomias_dinamicas.thy. Primero la primera (Crítica de la razón pura, A426-433 /
   B454-461), solo en su parte del tiempo; la del espacio tiene la misma forma.
   - Tesis: el mundo tiene un comienzo en el tiempo. Kant: si no lo tuviera, hasta cada instante habría transcurrido una
     serie infinita, y una serie infinita no puede completarse por síntesis sucesiva. Aquí, esa «síntesis completa» se
     traduce como buena fundación de la serie de los estados pasados del mundo (interpretación; la mesa lo dirá).
   - Antítesis: el mundo no tiene comienzo. Kant: si lo tuviera, lo precedería un tiempo vacío, y en un tiempo vacío
     ningún instante se distingue de otro como condición para que algo empiece. Aquí: el tiempo no tiene primer instante
     y el tiempo vacío es homogéneo.
   - Solución de Kant para las matemáticas (A501-507 / B529-535): tesis y antítesis no son contradictorias sino
     contrarias, como «todo cuerpo huele bien» y «todo cuerpo huele mal»: las dos son falsas si los cuerpos no huelen.
     Las dos presuponen que el mundo es un todo dado en el tiempo; con esa presuposición, una de las dos vale; sin ella,
     ninguna.
   Pruebas paso a paso, sin auto ni presburger sobre aritmética. *)

section ‹Primera antinomia: el comienzo del mundo en el tiempo›

typedecl instante
consts Antes :: "instante ⇒ instante ⇒ bool"   (* Antes s t: el instante s es anterior a t *)
consts Mundo :: "instante ⇒ bool"                (* Mundo t: el mundo (un estado del mundo) existe en t *)

(* El mundo empieza: hay un instante en que existe y antes del cual no existía. *)
definition comienzo :: bool where "comienzo ≡ ∃t0. Mundo t0 ∧ (∀s. Antes s t0 ⟶ ¬ Mundo s)"
(* El mundo existe y no tiene comienzo: antes de cada estado del mundo hubo otro. *)
definition sin_comienzo :: bool where
  "sin_comienzo ≡ (∃t. Mundo t) ∧ (∀t. Mundo t ⟶ (∃s. Antes s t ∧ Mundo s))"

(* Premisa propia de la tesis: la serie de los estados pasados del mundo está completa (sin regreso al infinito). *)
definition T1_sintesis_completa :: bool where
  "T1_sintesis_completa ≡ wf {(s, t). Antes s t ∧ Mundo s ∧ Mundo t}"
(* Premisas propias de la antítesis: el tiempo no tiene primer instante, y el tiempo vacío es homogéneo: si el mundo
   existe en t y antes de t no había nada, nada distingue a t de un instante vacío anterior s, y el mundo tendría que
   existir también en s. *)
definition A1_tiempo_sin_principio :: bool where "A1_tiempo_sin_principio ≡ ∀t. ∃s. Antes s t"
definition A1_tiempo_vacio_homogeneo :: bool where
  "A1_tiempo_vacio_homogeneo ≡ ∀t s. Mundo t ∧ (∀u. Antes u t ⟶ ¬ Mundo u) ∧ Antes s t ⟶ Mundo s"

(* Tesis: con la síntesis completa, si el mundo existe en algún instante, tiene un comienzo. *)
theorem tesis1:
  assumes "T1_sintesis_completa" "∃t. Mundo t"
  shows "comienzo"
proof -
  let ?r = "{(s, t). Antes s t ∧ Mundo s ∧ Mundo t}" and ?S = "{t. Mundo t}"
  obtain a where "Mundo a" using assms(2) by blast
  then have "a ∈ ?S" by simp
  have "∃m∈?S. ∀s. (s, m) ∈ ?r ⟶ s ∉ ?S"
    using wf_eq_minimal[THEN iffD1, OF assms(1)[unfolded T1_sintesis_completa_def], rule_format, OF ‹a ∈ ?S›] .
  then obtain m where m: "m ∈ ?S" and min: "∀s. (s, m) ∈ ?r ⟶ s ∉ ?S" by blast
  have mm: "Mundo m" using m by simp
  have "∀s. Antes s m ⟶ ¬ Mundo s"
  proof (intro allI impI notI)
    fix s assume "Antes s m" "Mundo s"
    then have "(s, m) ∈ ?r" using mm by simp
    then have "s ∉ ?S" using min by blast
    then show False using ‹Mundo s› by simp
  qed
  then show ?thesis unfolding comienzo_def using mm by blast
qed

(* Antítesis: si el tiempo no tiene principio y el tiempo vacío es homogéneo, el mundo no puede empezar. *)
theorem antitesis1:
  assumes "A1_tiempo_sin_principio" "A1_tiempo_vacio_homogeneo"
  shows "¬ comienzo"
proof
  assume "comienzo"
  then obtain t0 where e: "Mundo t0" and v: "∀s. Antes s t0 ⟶ ¬ Mundo s" unfolding comienzo_def by blast
  obtain s where a: "Antes s t0" using assms(1) unfolding A1_tiempo_sin_principio_def by blast
  have "Mundo s" using assms(2) e v a unfolding A1_tiempo_vacio_homogeneo_def by blast
  moreover have "¬ Mundo s" using v a by blast
  ultimately show False by contradiction
qed

(* Si el mundo existe como un todo en el tiempo, empezar y no empezar son contradictorios: vale uno de los dos. *)
theorem con_mundo_dado_contradictorias:
  assumes "∃t. Mundo t"
  shows "(comienzo ∨ sin_comienzo) ∧ ¬ (comienzo ∧ sin_comienzo)"
proof (rule conjI)
  show "comienzo ∨ sin_comienzo"
  proof (cases "∃m. Mundo m ∧ (∀s. Antes s m ⟶ ¬ Mundo s)")
    case True then show ?thesis unfolding comienzo_def by blast
  next
    case False
    then have "∀t. Mundo t ⟶ (∃s. Antes s t ∧ Mundo s)" by blast
    then show ?thesis unfolding sin_comienzo_def using assms by blast
  qed
  show "¬ (comienzo ∧ sin_comienzo)" unfolding comienzo_def sin_comienzo_def by blast
qed

(* Por eso, con el mundo dado, la antítesis da la otra mitad: sin comienzo. *)
corollary antitesis1_sin_comienzo:
  assumes "A1_tiempo_sin_principio" "A1_tiempo_vacio_homogeneo" "∃t. Mundo t"
  shows "sin_comienzo"
  using con_mundo_dado_contradictorias[OF assms(3)] antitesis1[OF assms(1) assms(2)] by blast

(* El choque: las premisas de la tesis y las de la antítesis, con el mundo dado, no caben juntas. *)
theorem choque1:
  assumes "T1_sintesis_completa" "A1_tiempo_sin_principio" "A1_tiempo_vacio_homogeneo" "∃t. Mundo t"
  shows False
  using tesis1[OF assms(1) assms(4)] antitesis1[OF assms(2) assms(3)] by blast

(* La solución de Kant: sin la presuposición de que el mundo es un todo dado en el tiempo, tesis y antítesis son
   contrarias y caen las dos (como «todo cuerpo huele bien» y «todo cuerpo huele mal» si los cuerpos no huelen). *)
theorem sin_mundo_dado_contrarias:
  assumes "¬ (∃t. Mundo t)"
  shows "¬ comienzo ∧ ¬ sin_comienzo"
  using assms unfolding comienzo_def sin_comienzo_def by blast

(* Cada lado tiene su modelo, con los enteros como instantes.
   Tesis: el mundo existe desde el 0; la serie de sus estados pasados está bien fundada y empieza en 0.
   (La homogeneidad del tiempo vacío falla: el -1 está vacío y el 0 no.) *)
lemma modelo_tesis1:
  fixes M :: "int ⇒ bool"
  defines "M ≡ λt. 0 ≤ t"
  shows "wf {(s, t). s < t ∧ M s ∧ M t} ∧ (∃t0. M t0 ∧ (∀s. s < t0 ⟶ ¬ M s))"
proof (rule conjI)
  have "{(s, t). s < t ∧ M s ∧ M t} ⊆ inv_image less_than nat"
  proof (rule subsetI)
    fix p assume "p ∈ {(s, t). s < t ∧ M s ∧ M t}"
    then obtain s t where p: "p = (s, t)" and st: "s < t" and s0: "0 ≤ s" unfolding M_def by blast
    have "nat s < nat t" using st s0 by (simp add: nat_less_eq_zless)
    then show "p ∈ inv_image less_than nat" unfolding p by simp
  qed
  then show "wf {(s, t). s < t ∧ M s ∧ M t}" by (rule wf_subset [OF wf_inv_image [OF wf_less_than]])
  show "∃t0. M t0 ∧ (∀s. s < t0 ⟶ ¬ M s)"
  proof (rule exI [of _ 0])
    show "M 0 ∧ (∀s. s < 0 ⟶ ¬ M s)" unfolding M_def by simp
  qed
qed

(* Antítesis: el mundo existe siempre; el tiempo no tiene principio, la homogeneidad se cumple (nunca hay un instante
   con el mundo y nada antes) y no hay comienzo; la síntesis completa falla: hay regreso sin fin. *)
lemma modelo_antitesis1:
  fixes M :: "int ⇒ bool"
  defines "M ≡ λt. True"
  shows "(∀t::int. ∃s. s < t) ∧ (∀t s. M t ∧ (∀u. u < t ⟶ ¬ M u) ∧ s < t ⟶ M s)
         ∧ ¬ (∃t0. M t0 ∧ (∀s. s < t0 ⟶ ¬ M s)) ∧ ¬ wf {(s, t). s < t ∧ M s ∧ M t}"
proof (intro conjI)
  show "∀t::int. ∃s. s < t"
  proof
    fix t :: int
    show "∃s. s < t" by (rule exI [of _ "t - 1"]) simp
  qed
  show "∀t s. M t ∧ (∀u. u < t ⟶ ¬ M u) ∧ s < t ⟶ M s" unfolding M_def by simp
  show "¬ (∃t0. M t0 ∧ (∀s. s < t0 ⟶ ¬ M s))"
  proof
    assume "∃t0. M t0 ∧ (∀s. s < t0 ⟶ ¬ M s)"
    then obtain t0 where "∀s. s < t0 ⟶ ¬ M s" by blast
    then have "¬ M (t0 - 1)" by simp
    then show False unfolding M_def by simp
  qed
  show "¬ wf {(s, t). s < t ∧ M s ∧ M t}"
  proof
    assume w: "wf {(s, t). s < t ∧ M s ∧ M t}"
    have h: "∀i. (- int (Suc i), - int i) ∈ {(s, t). s < t ∧ M s ∧ M t}" unfolding M_def by simp
    have "∃f. ∀i. (f (Suc i), f i) ∈ {(s, t). s < t ∧ M s ∧ M t}"
    proof (rule exI [of _ "λn. - int n"])
      show "∀i. ((λn. - int n) (Suc i), (λn. - int n) i) ∈ {(s, t). s < t ∧ M s ∧ M t}" using h by simp
    qed
    then show False using w unfolding wf_iff_no_infinite_down_chain by blast
  qed
qed

section ‹Segunda antinomia: lo simple (A434-443 / B462-471)›

(* - Tesis: toda sustancia compuesta consta de partes simples. Kant: la composición es solo una relación accidental
     entre sustancias; si se suprimiera toda composición, algo tendría que quedar, y eso serían partes simples. Aquí:
     la división en partes acaba (la relación «ser parte propia de» está bien fundada), más la transitividad.
   - Antítesis: nada en el mundo es simple. Kant: todo lo compuesto ocupa un espacio, el espacio no consta de partes
     simples sino de espacios, y lo que ocupa un espacio tiene partes unas fuera de otras. Aquí: lo que está en el
     espacio tiene partes que también están en el espacio.
   - Diferencia con la primera: aquí tesis y antítesis no son contradictorias ni siquiera dando por hecho que hay
     compuestos. Un mundo mixto (algo simple y un compuesto sin partes simples) las hace falsas a las dos: son
     contrarias, como dice Kant. Su razón es que la división de un fenómeno solo va «al indefinido»: ni se llega a
     partes simples ni están dadas infinitas partes (A523-527 / B551-555); eso ya no es lógica, y la mesa lo dirá. *)

typedecl porcion
consts Parte :: "porcion ⇒ porcion ⇒ bool"   (* Parte y x: y es una parte propia de x *)
consts EnEspacio :: "porcion ⇒ bool"            (* x ocupa un espacio: es un fenómeno *)

definition simple :: "porcion ⇒ bool" where "simple x ≡ ¬ (∃y. Parte y x)"
definition compuesto :: "porcion ⇒ bool" where "compuesto x ≡ ∃y. Parte y x"

(* Premisa de todo el mundo: la parte de una parte es parte. *)
definition M_transitiva :: bool where "M_transitiva ≡ ∀x y z. Parte x y ∧ Parte y z ⟶ Parte x z"
(* Premisa propia de la tesis: la división acaba, no hay partes de partes sin fin. *)
definition T2_division_acaba :: bool where "T2_division_acaba ≡ wf {(y, x). Parte y x}"
(* Premisa propia de la antítesis: lo que está en el espacio tiene partes que están en el espacio. *)
definition A2_espacio_divisible :: bool where
  "A2_espacio_divisible ≡ ∀x. EnEspacio x ⟶ (∃y. Parte y x ∧ EnEspacio y)"

(* Tesis: si la división acaba, todo compuesto tiene alguna parte simple. *)
theorem tesis2:
  assumes "T2_division_acaba" "M_transitiva"
  shows "∀x. compuesto x ⟶ (∃s. simple s ∧ Parte s x)"
proof (intro allI impI)
  fix x assume "compuesto x"
  then obtain y0 where "Parte y0 x" unfolding compuesto_def by blast
  let ?r = "{(y, x). Parte y x}" and ?S = "{y. Parte y x}"
  have "y0 ∈ ?S" using ‹Parte y0 x› by simp
  have "∃m∈?S. ∀z. (z, m) ∈ ?r ⟶ z ∉ ?S"
    using wf_eq_minimal[THEN iffD1, OF assms(1)[unfolded T2_division_acaba_def], rule_format, OF ‹y0 ∈ ?S›] .
  then obtain m where m: "m ∈ ?S" and min: "∀z. (z, m) ∈ ?r ⟶ z ∉ ?S" by blast
  have pm: "Parte m x" using m by simp
  have "simple m" unfolding simple_def
  proof
    assume "∃z. Parte z m"
    then obtain z where zm: "Parte z m" by blast
    then have "Parte z x" using pm assms(2) unfolding M_transitiva_def by blast
    then have "z ∈ ?S" by simp
    moreover have "(z, m) ∈ ?r" using zm by simp
    ultimately show False using min by blast
  qed
  then show "∃s. simple s ∧ Parte s x" using pm by blast
qed

(* Antítesis: si lo que está en el espacio es divisible en el espacio, nada en el espacio es simple. *)
theorem antitesis2:
  assumes "A2_espacio_divisible"
  shows "¬ (∃s. EnEspacio s ∧ simple s)"
  using assms unfolding A2_espacio_divisible_def simple_def by blast

(* El choque: si algo ocupa un espacio, la división que acaba y el espacio divisible no caben juntos. *)
theorem choque2:
  assumes "T2_division_acaba" "A2_espacio_divisible" "∃x. EnEspacio x"
  shows False
proof -
  let ?r = "{(y, x). Parte y x}" and ?S = "{x. EnEspacio x}"
  obtain a where "EnEspacio a" using assms(3) by blast
  then have "a ∈ ?S" by simp
  have "∃m∈?S. ∀z. (z, m) ∈ ?r ⟶ z ∉ ?S"
    using wf_eq_minimal[THEN iffD1, OF assms(1)[unfolded T2_division_acaba_def], rule_format, OF ‹a ∈ ?S›] .
  then obtain m where m: "m ∈ ?S" and min: "∀z. (z, m) ∈ ?r ⟶ z ∉ ?S" by blast
  have "EnEspacio m" using m by simp
  then obtain y where "Parte y m" and "EnEspacio y" using assms(2) unfolding A2_espacio_divisible_def by blast
  then show False using min by simp
qed

(* Modelo de la tesis: los naturales, con «y es parte de x» si y < x. El 0 es simple y es parte de todo compuesto. *)
lemma modelo_tesis2:
  fixes P :: "nat ⇒ nat ⇒ bool"
  defines "P ≡ λy x. y < x"
  shows "wf {(y, x). P y x} ∧ (∀x y z. P x y ∧ P y z ⟶ P x z)
         ∧ (∀x. (∃y. P y x) ⟶ (∃s. ¬ (∃y. P y s) ∧ P s x))"
proof (intro conjI)
  show "wf {(y, x). P y x}" unfolding P_def by (rule wf_less)
  show "∀x y z. P x y ∧ P y z ⟶ P x z" unfolding P_def by simp
  show "∀x. (∃y. P y x) ⟶ (∃s. ¬ (∃y. P y s) ∧ P s x)"
  proof (intro allI impI)
    fix x assume "∃y. P y x"
    then obtain y where "y < x" unfolding P_def by blast
    then have "0 < x" by simp
    show "∃s. ¬ (∃y. P y s) ∧ P s x"
    proof (rule exI [of _ 0])
      show "¬ (∃y. P y 0) ∧ P 0 x" unfolding P_def using ‹0 < x› by simp
    qed
  qed
qed

(* Modelo de la antítesis: los naturales, con «y es parte de x» si x < y (cada pieza se parte en otra más pequeña,
   x + 1, sin fin). Todo está en el espacio, nada es simple y la división no acaba. *)
(* Con la premisa de la tesis, la otra salida (todo divisible sin fin) es imposible: si la división acaba, algo es
   simple. Es lo que soltaría quien, como Schopenhauer, solo da por buenas las pruebas de las antítesis. *)
lemma division_acaba_hay_simple:
  assumes "T2_division_acaba"
  shows "∃x. simple x"
proof -
  have "∃m∈UNIV. ∀y. (y, m) ∈ {(y, x). Parte y x} ⟶ y ∉ UNIV"
    using wf_eq_minimal[THEN iffD1, OF assms[unfolded T2_division_acaba_def], rule_format, OF UNIV_I] .
  then obtain m where "∀y. (y, m) ∈ {(y, x). Parte y x} ⟶ y ∉ UNIV" by blast
  then have "simple m" unfolding simple_def by simp
  then show ?thesis by blast
qed

lemma modelo_antitesis2:
  fixes P :: "nat ⇒ nat ⇒ bool"
  defines "P ≡ λy x. x < y"
  shows "(∀x y z. P x y ∧ P y z ⟶ P x z) ∧ (∀x. ∃y. P y x) ∧ ¬ wf {(y, x). P y x}"
proof (intro conjI)
  show "∀x y z. P x y ∧ P y z ⟶ P x z" unfolding P_def by simp
  show "∀x. ∃y. P y x"
  proof
    fix x :: nat
    show "∃y. P y x" unfolding P_def by (rule exI [of _ "Suc x"]) simp
  qed
  show "¬ wf {(y, x). P y x}"
  proof
    assume w: "wf {(y, x). P y x}"
    have "∃f. ∀i. (f (Suc i), f i) ∈ {(y, x). P y x}"
    proof (rule exI [of _ "λn. n"])
      show "∀i. ((λn. n) (Suc i), (λn. n) i) ∈ {(y, x). P y x}" unfolding P_def by simp
    qed
    then show False using w unfolding wf_iff_no_infinite_down_chain by blast
  qed
qed

(* Contrarias: un mundo mixto en que las dos fallan. El 0 es simple (la antítesis falla) y el 1 es compuesto sin
   ninguna parte simple (la tesis falla): por encima del 0, cada pieza x se parte en x + 1 sin fin. *)
lemma mixto_contrarias:
  fixes P :: "nat ⇒ nat ⇒ bool"
  defines "P ≡ λy x. 0 < x ∧ x < y"
  shows "(∀x y z. P x y ∧ P y z ⟶ P x z) ∧ (∃s. ¬ (∃y. P y s))
         ∧ (∃x. (∃y. P y x) ∧ ¬ (∃s. ¬ (∃y. P y s) ∧ P s x))"
proof (intro conjI)
  show "∀x y z. P x y ∧ P y z ⟶ P x z" unfolding P_def by simp
  show "∃s. ¬ (∃y. P y s)"
  proof (rule exI [of _ 0])
    show "¬ (∃y. P y 0)" unfolding P_def by simp
  qed
  show "∃x. (∃y. P y x) ∧ ¬ (∃s. ¬ (∃y. P y s) ∧ P s x)"
  proof (rule exI [of _ 1], rule conjI)
    show "∃y. P y 1"
    proof (rule exI [of _ 2])
      show "P 2 1" unfolding P_def by simp
    qed
    show "¬ (∃s. ¬ (∃y. P y s) ∧ P s 1)"
    proof
      assume "∃s. ¬ (∃y. P y s) ∧ P s 1"
      then obtain s where ns: "¬ (∃y. P y s)" and p: "P s 1" by blast
      from p have "1 < s" unfolding P_def by simp
      then have "P (Suc s) s" unfolding P_def by simp
      then show False using ns by blast
    qed
  qed
qed
