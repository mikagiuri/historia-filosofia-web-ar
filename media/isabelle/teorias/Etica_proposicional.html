Theory Etica_proposicional

theory Etica_proposicional
imports Main
begin

text ‹La Ética entera, en argumentación proposicional — ahora solo con negación,
  disyunción y conjunción (¬, ∨, ∧). Se prescinde del condicional y del
  bicondicional mediante sus equivalentes:

    A ⟶ B  ≡  ¬ A ∨ B                (el condicional es una disyunción)
    A ⟷ B  ≡  (A ∧ B) ∨ (¬ A ∧ ¬ B)   («o ambos o ninguno»)

  Todo se demuestra por pura lógica proposicional (blast), sin ⟶ ni
  ⟷ ni cuantificadores.›

consts
  (* Parte I *)
  Eficaz :: bool        (* P36: todo es eficaz *)
  Contingencia :: bool  (* P29: hay algo contingente (se niega) *)
  (* Parte II *)
  Identidad :: bool     (* lectura de identidad (II.7) *)
  Paralelismo :: bool   (* el orden de las ideas = el orden de las cosas *)
  Adecuacion :: bool    (* II.43: la verdad es la adecuación *)
  (* Parte III *)
  Conato :: bool        (* III.6: todo persevera en su ser *)
  Actividad :: bool     (* III.1-3: obrar = causa adecuada *)
  Alegria :: bool       (* III.11 *)
  Amor :: bool          (* III.13 *)
  (* Parte IV *)
  Virtud :: bool        (* IV.8: virtud = potencia *)
  Bueno :: bool         (* def.1 *)
  Malo :: bool          (* def.2 *)
  (* Parte V *)
  Beatitud :: bool      (* V.42: beatitud = virtud *)

axiomatization where
  A_identidad:        "¬ Identidad ∨ Paralelismo" and
  A_no_contingencia:  "¬ Contingencia" and
  A_p36:              "Eficaz" and
  A_conato_es_potencia:   "(Conato ∧ Eficaz) ∨ (¬ Conato ∧ ¬ Eficaz)" and
  A_adecuacion_actividad: "(Actividad ∧ Adecuacion) ∨ (¬ Actividad ∧ ¬ Adecuacion)" and
  A_virtud_actividad:     "(Virtud ∧ Actividad) ∨ (¬ Virtud ∧ ¬ Actividad)" and
  A_virtud_es_conato: "¬ Virtud ∨ Conato" and
  A_beatitud_virtud:  "(Beatitud ∧ Virtud) ∨ (¬ Beatitud ∧ ¬ Virtud)" and
  A_amor_es_alegria:  "¬ Amor ∨ Alegria" and
  A_bueno_malo:       "¬ Bueno ∨ ¬ Malo"

text ‹III.6 — todo persevera. El «modus ponens» se vuelve silogismo disyuntivo:
  de «Eficaz» y de «Conato ∨ ¬ Eficaz» (la mitad del bicondicional) sale
  «Conato».›
theorem III6: "Conato"
  using A_p36 A_conato_es_potencia by blast

text ‹La cadena de la libertad (II ⟶ V) se comprime en un solo bicondicional:
  adecuación y beatitud «o ambas o ninguna».›
theorem libertad_es_beatitud:
  "(Adecuacion ∧ Beatitud) ∨ (¬ Adecuacion ∧ ¬ Beatitud)"
  using A_adecuacion_actividad A_virtud_actividad A_beatitud_virtud by blast

text ‹La beatitud es una forma del conato: «no-beatitud o conato» (pero no al
  revés: no todo lo que persevera es libre).›
theorem beatitud_implica_conato: "¬ Beatitud ∨ Conato"
  using A_beatitud_virtud A_virtud_es_conato by blast

text ‹El amor es una forma de la alegría.›
theorem amor_presupone_alegria: "¬ Amor ∨ Alegria"
  using A_amor_es_alegria by blast

text ‹El bien excluye el mal: no puede darse «bueno y malo» a la vez.›
theorem bueno_no_malo: "¬ (Bueno ∧ Malo)"
  using A_bueno_malo by blast

text ‹Las dos cadenas del libro, en un solo enunciado (solo ¬, ∨, ∧):›
theorem las_dos_cadenas:
  "((Conato ∧ Eficaz) ∨ (¬ Conato ∧ ¬ Eficaz)) ∧
   ((Adecuacion ∧ Beatitud) ∨ (¬ Adecuacion ∧ ¬ Beatitud)) ∧
   (¬ Beatitud ∨ Conato) ∧ (¬ Contingencia)"
proof -
  have c1: "(Conato ∧ Eficaz) ∨ (¬ Conato ∧ ¬ Eficaz)" using A_conato_es_potencia by blast
  have c2: "(Adecuacion ∧ Beatitud) ∨ (¬ Adecuacion ∧ ¬ Beatitud)"
    using A_adecuacion_actividad A_virtud_actividad A_beatitud_virtud by blast
  have c3: "¬ Beatitud ∨ Conato" using A_beatitud_virtud A_virtud_es_conato by blast
  have c4: "¬ Contingencia" using A_no_contingencia by blast
  from c1 c2 c3 c4 show ?thesis by blast
qed

end
