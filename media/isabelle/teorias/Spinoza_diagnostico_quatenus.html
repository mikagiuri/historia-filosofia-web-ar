Theory Spinoza_diagnostico_quatenus

theory Spinoza_diagnostico_quatenus
  imports Spinoza_Etica_I
begin

(* Diagnóstico: ¿qué premisa impide el mundo con identidad de cuerpo e idea? *)

abbreviation identidad :: bool where
  "identidad ≡ ∃c1 c2. c1 ≠ c2 ∧ ModoDeAtributo c1 Extension ∧ ModoDeAtributo c1 Pensamiento
        ∧ ModoDeAtributo c2 Extension ∧ ModoDeAtributo c2 Pensamiento
        ∧ CausaBajo Extension Extension c1 c2 ∧ IdeaDe c1 c1 ∧ IdeaDe c2 c2"

abbreviation base_I :: bool where
  "base_I ≡ A1_excluyente ∧ H_por_si_excluye_por_otro ∧ A2 ∧ A4 ∧ A5 ∧ A7 ∧ D1 ∧ D4 ∧ D6
   ∧ H_conversa_A5 ∧ H_quien_comparte_uno_comparte_todos ∧ H_individuacion_por_atributos
   ∧ H_ser_en_implica_concebirse ∧ H_solo_en_si_solo_por_si ∧ H_atributo_compartido_es_comun
   ∧ H_todo_tiene_causa ∧ H_sustancia_con_atributo ∧ H_modo_en_sustancia ∧ H_inherencia_es_causalidad"

abbreviation modos_inf :: bool where
  "modos_inf ≡ H_modo_de_atributo_es_modo ∧ H_todo_modo_de_algun_atributo ∧ H_limitar_es_determinar
   ∧ H_lo_que_limita_es_finito ∧ H_absoluto_solo_de_Dios ∧ H_mediato_de_Dios_y_su_modificacion
   ∧ H_dos_maneras_de_seguirse ∧ H_de_lo_finito_lo_finito"

abbreviation mente_cuerpo :: bool where
  "mente_cuerpo ≡ H_atributos_distintos ∧ H_modo_se_concibe_por_su_atributo
   ∧ H_paralelismo ∧ H_ideas_del_pensamiento ∧ H_concebir_es_causar_ideas"

abbreviation quatenus :: bool where
  "quatenus ≡ A4_quatenus ∧ H_atributo_por_si_quatenus ∧ H_en_cuanto_requiere_atributo
   ∧ H_mismo_orden ∧ H_causa_relativa_es_causa"

lemma solo_quatenus: "quatenus ∧ H_atributos_distintos ∧ identidad"
  nitpick [satisfy, timeout = 120] oops

lemma quatenus_mente_cuerpo: "quatenus ∧ mente_cuerpo ∧ identidad"
  nitpick [satisfy, timeout = 120] oops

lemma quatenus_base: "quatenus ∧ H_atributos_distintos ∧ base_I ∧ identidad"
  nitpick [satisfy, timeout = 120] oops

lemma quatenus_modos: "quatenus ∧ H_atributos_distintos ∧ modos_inf ∧ identidad"
  nitpick [satisfy, timeout = 120] oops

lemma quatenus_base_modos: "quatenus ∧ H_atributos_distintos ∧ base_I ∧ modos_inf ∧ identidad"
  nitpick [satisfy, timeout = 120] oops

lemma quatenus_base_mente: "quatenus ∧ base_I ∧ mente_cuerpo ∧ identidad"
  nitpick [satisfy, timeout = 120] oops

end
