Theory Mundo_tiempo_definido

theory Mundo_tiempo_definido
  imports Main
begin

(* Tercer mundo del Laboratorio (21-10-2026, propuesta del profesor): «El tiempo, definido».
   El mundo de práctica dejaba Antes y DuraMas como relaciones sueltas. Aquí se dice qué es el tiempo:
   instantes ordenados (los números naturales: el tiempo tiene un principio) y, para cada cosa, el instante
   en que empieza y el instante en que acaba. Antes, DuraMas, Sobrevive y ExisteEn pasan a ser definiciones;
   Crea sigue siendo primitiva. Mismos nombres que la mesa web. *)

typedecl cosa

consts
  empieza :: "cosa ⇒ nat"
  acaba :: "cosa ⇒ nat"
  Crea :: "cosa ⇒ cosa ⇒ bool"

(* Lo que es el tiempo en este mundo: nada acaba antes de empezar. *)
definition Tiempo :: bool where "Tiempo ≡ ∀x. empieza x ≤ acaba x"

definition Antes :: "cosa ⇒ cosa ⇒ bool" where "Antes x y ≡ empieza x < empieza y"
definition Sobrevive :: "cosa ⇒ cosa ⇒ bool" where "Sobrevive x y ≡ acaba y < acaba x"
definition DuraMas :: "cosa ⇒ cosa ⇒ bool" where "DuraMas x y ≡ acaba y - empieza y < acaba x - empieza x"
definition ExisteEn :: "cosa ⇒ nat ⇒ bool" where "ExisteEn x t ≡ empieza x ≤ t ∧ t ≤ acaba x"
definition increado :: "cosa ⇒ bool" where "increado x ≡ ¬ (∃y. Crea y x)"

(* Premisas de la creación: estas sí se pueden quitar. *)
definition K_existe :: bool where "K_existe ≡ ∀x y. Crea x y ⟶ ExisteEn x (empieza y)"
definition K_antes :: bool where "K_antes ≡ ∀x y. Crea x y ⟶ Antes x y"
definition K_no_despues :: bool where "K_no_despues ≡ ∀x y. Crea x y ⟶ empieza x ≤ empieza y"
definition K_un_creador :: bool where "K_un_creador ≡ ∀x y z. Crea x z ∧ Crea y z ⟶ x = y"
definition K_todo_creado :: bool where "K_todo_creado ≡ ∀x. ∃y. Crea y x"

(* R1-R3: lo que en el mundo de práctica eran premisas, aquí son teoremas, sin premisa ninguna. *)
theorem R1: "¬ Antes x x" unfolding Antes_def by simp
theorem R2: "Antes x y ∧ Antes y z ⟶ Antes x z" unfolding Antes_def by simp
theorem R3: "Antes x y ⟶ ¬ Antes y x" unfolding Antes_def by simp

(* R4: Nada se crea a sí mismo. *)
theorem R4:
  assumes "K_antes"
  shows "¬ Crea x x"
  using assms R1 unfolding K_antes_def by blast

lemma R4_sin_premisa: "¬ Crea x x"
  nitpick [expect = genuine, timeout = 120]
  oops

(* R5 y R6: posibilidades. Un autor (True) que empieza en 0 y acaba en 2 crea en el instante 1 un libro
   (False) que acaba en 3: el libro sobrevive al autor. Con el libro hasta 5, además dura más. *)
lemma R5_posible:
  fixes e a :: "bool ⇒ nat" and C :: "bool ⇒ bool ⇒ bool"
  defines "e ≡ λx. if x then 0 else 1" and "a ≡ λx. if x then 2 else 3" and "C ≡ λx y. x ∧ ¬ y"
  shows "(∀x. e x ≤ a x) ∧ (∀x y. C x y ⟶ e x ≤ e y ∧ e y ≤ a x) ∧ (∀x y. C x y ⟶ e x < e y)
         ∧ (∀x y z. C x z ∧ C y z ⟶ x = y) ∧ (∃x y. C x y ∧ a x < a y)"
  unfolding e_def a_def C_def by auto

lemma R6_posible:
  fixes e a :: "bool ⇒ nat" and C :: "bool ⇒ bool ⇒ bool"
  defines "e ≡ λx. if x then 0 else 1" and "a ≡ λx. if x then 2 else 5" and "C ≡ λx y. x ∧ ¬ y"
  shows "(∀x. e x ≤ a x) ∧ (∀x y. C x y ⟶ e x ≤ e y ∧ e y ≤ a x) ∧ (∀x y. C x y ⟶ e x < e y)
         ∧ (∀x y z. C x z ∧ C y z ⟶ x = y) ∧ (∃x y. C x y ∧ a x - e x < a y - e y)"
  unfolding e_def a_def C_def by auto

(* R7: Lo que sobrevive dura más. No se sigue: se puede empezar tarde y acabar un poco después. *)
lemma R7_no_se_sigue: "Tiempo ⟹ Sobrevive x y ⟶ DuraMas x y"
  nitpick [expect = genuine, timeout = 120]
  oops

(* R8: Quien empieza antes y acaba después dura más. Se sigue de lo que es el tiempo. *)
theorem R8:
  assumes "Tiempo"
  shows "Antes x y ∧ Sobrevive x y ⟶ DuraMas x y"
proof
  assume h: "Antes x y ∧ Sobrevive x y"
  have "empieza y ≤ acaba y" using assms unfolding Tiempo_def by blast
  then show "DuraMas x y" using h unfolding Antes_def Sobrevive_def DuraMas_def by linarith
qed

lemma R8_sin_tiempo: "Antes x y ∧ Sobrevive x y ⟶ DuraMas x y"
  nitpick [expect = genuine, timeout = 120]
  oops

(* R9: Hay algo increado. Como el tiempo tiene un principio (los naturales), lo que empieza primero no puede
   tener creador si quien crea empezó antes. Vale con cualquier número de cosas, también infinito:
   es el contraste con el mundo de práctica, donde solo valía en los mundos finitos. *)
theorem R9:
  assumes "K_antes"
  shows "∃x. increado x"
proof -
  obtain x where min: "∀y. empieza x ≤ empieza y"
  proof -
    let ?n = "LEAST n. ∃z. empieza z = n"
    have "∃z. empieza z = ?n" by (rule LeastI_ex) blast
    then obtain z where z: "empieza z = ?n" by blast
    have "∀y. empieza z ≤ empieza y" using z by (metis (mono_tags) Least_le)
    then show ?thesis using that by blast
  qed
  have "¬ (∃y. Crea y x)" using assms min unfolding K_antes_def Antes_def by (meson leD)
  then show ?thesis unfolding increado_def by blast
qed

(* Y por eso «todo ha sido creado» es incompatible con «quien crea empezó antes», aquí sin necesidad de que el mundo sea finito. *)
corollary todo_creado_incompatible:
  assumes "K_antes"
  shows "¬ K_todo_creado"
  using R9 [OF assms] unfolding increado_def K_todo_creado_def by blast

(* La 9 no obliga a aceptar algo increado: dice que no caben juntas tres cosas (todo tiene creador, quien crea empezó
   antes y el tiempo tiene principio). Observación del profesor (21-10): quien quiera que todo tenga creador puede renunciar
   a K_antes, y admitir que se crea A LA VEZ o que algo se crea a sí mismo (la causa sui de Spinoza, Ética I, def. 1).
   R10: «Todo puede tener creador». Con K_no_despues (crear a la vez), K_existe y K_un_creador, una sola cosa que se crea a sí misma. *)
lemma R10_posible:
  fixes e a :: "bool ⇒ nat" and C :: "bool ⇒ bool ⇒ bool"
  defines "e ≡ λx. 0" and "a ≡ λx. 1" and "C ≡ λx y. x = y"
  shows "(∀x. e x ≤ a x) ∧ (∀x y. C x y ⟶ e x ≤ e y ∧ e y ≤ a x) ∧ (∀x y. C x y ⟶ e x ≤ e y)
         ∧ (∀x y z. C x z ∧ C y z ⟶ x = y) ∧ (∀x. ∃y. C y x)"
  unfolding e_def a_def C_def by auto

(* Y con K_antes no es posible: es la 9 (todo_creado_incompatible). *)

end
