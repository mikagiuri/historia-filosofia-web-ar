Theory Spinoza_Etica_I

(*  Spinoza, Ethica more geometrico demonstrata, Parte I (def. 1-8, ax. 1-7, prop. 1-19, 21-24, 26, 28, 29, 32 y 36)
    Reconstrucción en Isabelle/HOL.

    Idea: cada axioma de Spinoza es una constante booleana (A1, A2...) y cada
    premisa que Spinoza NO enuncia pero necesita es otra (H_...). Así cada
    teorema muestra exactamente de qué depende, y Nitpick (buscador de
    contramodelos) enseña qué pasa si se quita la premisa oculta.
*)

theory Spinoza_Etica_I
  imports Main
begin

section ‹Vocabulario primitivo›

typedecl cosa
typedecl atributo

consts
  En      :: "cosa ⇒ cosa ⇒ bool"      ― ‹En x y: x es en y (inest)›
  SeConcibePor :: "cosa ⇒ cosa ⇒ bool"      ― ‹SeConcibePor x y: x se concibe por y›
  Causa   :: "cosa ⇒ cosa ⇒ bool"      ― ‹Causa x y: x es causa de y›
  AlgoEnComun   :: "cosa ⇒ cosa ⇒ bool"      ― ‹AlgoEnComun x y: x e y tienen algo en común›
  TieneAtributo   :: "cosa ⇒ atributo ⇒ bool"  ― ‹TieneAtributo s a: a es atributo de s›


section ‹Definiciones›

text ‹Def. 3: «Por sustancia entiendo aquello que es en sí y se concibe por sí, esto es,
  aquello cuyo concepto no necesita del concepto de otra cosa». La cláusula final
  (no se concibe por ninguna otra cosa) es la que hace casi todo el trabajo.
  Versión endurecida: tampoco es en ninguna otra cosa. Sin esta cláusula, Nitpick
  construía sustancias que, además de ser en sí, eran en otra.›

definition sustancia :: "cosa ⇒ bool" where
  "sustancia x ≡ En x x ∧ SeConcibePor x x ∧ (∀y. En x y ⟶ y = x)
                   ∧ (∀y. SeConcibePor x y ⟶ y = x)"

text ‹Def. 5: «Por modo entiendo las afecciones de la sustancia, o sea, aquello que es
  en otra cosa, por medio de la cual también se concibe».›

definition modo :: "cosa ⇒ bool" where
  "modo x ≡ ∃y. y ≠ x ∧ En x y ∧ SeConcibePor x y"

definition afeccion :: "cosa ⇒ cosa ⇒ bool" where
  "afeccion m s ≡ m ≠ s ∧ En m s ∧ SeConcibePor m s"

text ‹«Anterior por naturaleza»: y se concibe por x, pero x no por y.›

definition anterior_por_naturaleza :: "cosa ⇒ cosa ⇒ bool" where
  "anterior_por_naturaleza x y ≡ SeConcibePor y x ∧ ¬ SeConcibePor x y"


section ‹Axiomas de Spinoza›

definition A1 :: bool where   ― ‹Todo lo que es, es en sí o en otra cosa.›
  "A1 ≡ ∀x. En x x ∨ (∃y. y ≠ x ∧ En x y)"

definition A2 :: bool where   ― ‹Lo que no puede concebirse por otra cosa, debe concebirse por sí.›
  "A2 ≡ ∀x. ¬ (∃y. y ≠ x ∧ SeConcibePor x y) ⟶ SeConcibePor x x"

definition A4 :: bool where   ― ‹El conocimiento del efecto depende del de la causa y lo implica.›
  "A4 ≡ ∀x y. Causa x y ⟶ SeConcibePor y x"

definition A5 :: bool where   ― ‹Las cosas que no tienen nada en común no se entienden una por otra.›
  "A5 ≡ ∀x y. ¬ AlgoEnComun x y ⟶ ¬ SeConcibePor x y ∧ ¬ SeConcibePor y x"


section ‹Premisas que Spinoza no enuncia›

text ‹H_conversa_A5: la conversa del axioma 5. Si dos cosas tienen algo en común,
  una se concibe por la otra. Spinoza la usa en la prop. 2 sin decirlo.›

definition H_conversa_A5 :: bool where
  "H_conversa_A5 ≡ ∀x y. AlgoEnComun x y ⟶ SeConcibePor x y ∨ SeConcibePor y x"

text ‹H_quien_comparte_uno_comparte_todos: si dos sustancias comparten un atributo, comparten todos.
  Es lo que Spinoza da por concedido en la prop. 5 y lo que Leibniz le objetó:
  ¿por qué no dos sustancias con un atributo común y otro distinto?›

definition H_quien_comparte_uno_comparte_todos :: bool where
  "H_quien_comparte_uno_comparte_todos ≡ ∀s t a. TieneAtributo s a ∧ TieneAtributo t a ⟶ (∀b. TieneAtributo s b ⟷ TieneAtributo t b)"

text ‹H_individuacion_por_atributos: lo que la prop. 4 más la prop. 1 le dan a Spinoza en la prop. 5.
  Dejadas aparte las afecciones, dos sustancias solo se distinguen por sus atributos.›

definition H_individuacion_por_atributos :: bool where
  "H_individuacion_por_atributos ≡ ∀s t. sustancia s ∧ sustancia t ∧ (∀b. TieneAtributo s b ⟷ TieneAtributo t b) ⟶ s = t"

text ‹H_ser_en_igual_concebirse: «ser en» y «concebirse por» son la misma relación.›

definition H_ser_en_igual_concebirse :: bool where
  "H_ser_en_igual_concebirse ≡ ∀x y. En x y ⟷ SeConcibePor x y"

text ‹Versión debilitada, solo para sustancias (se usa desde que la fuerte chocó con
  las partes y las causas de los modos; ver «Las partes de los modos»). Lo que es en
  otra cosa se concibe por ella, para todo; pero «concebirse por ⟹ ser en» solo se
  exige a lo que no es en nada más que en sí mismo. Un modo puede así concebirse por
  sus causas y sus partes sin ser en ellas.›

definition H_ser_en_implica_concebirse :: bool where
  "H_ser_en_implica_concebirse ≡ ∀x y. En x y ⟶ SeConcibePor x y"

definition H_solo_en_si_solo_por_si :: bool where
  "H_solo_en_si_solo_por_si ≡ ∀x. (∀y. En x y ⟶ y = x) ⟶ (∀y. SeConcibePor x y ⟶ En x y)"

lemma H_en_concibe_implica_debiles:
  "H_ser_en_igual_concebirse ⟹ H_ser_en_implica_concebirse ∧ H_solo_en_si_solo_por_si"
  unfolding H_ser_en_igual_concebirse_def H_ser_en_implica_concebirse_def H_solo_en_si_solo_por_si_def by blast

definition H_atributo_compartido_es_comun :: bool where   ― ‹Compartir un atributo es tener algo en común.›
  "H_atributo_compartido_es_comun ≡ ∀s t a. TieneAtributo s a ∧ TieneAtributo t a ⟶ AlgoEnComun s t"


section ‹Definición negativa de sustancia›

text ‹Lectura alternativa de la def. 3, más cercana al «esto es» de Spinoza:
  sustancia es lo que no es en otra cosa ni se concibe por otra cosa.›

definition sustancia_negativa :: "cosa ⇒ bool" where
  "sustancia_negativa x ≡ ¬ (∃y. y ≠ x ∧ En x y) ∧ ¬ (∃y. y ≠ x ∧ SeConcibePor x y)"

text ‹Con los ax. 1 y 2, las dos definiciones son equivalentes: el ax. 1 da el «es en
  sí» a lo que no es en otro, y el ax. 2 da el «se concibe por sí» a lo que no se
  concibe por otro.›

theorem sustancia_equiv_neg:
  assumes "A1" "A2"
  shows "sustancia x ⟷ sustancia_negativa x"
  using assms unfolding A1_def A2_def sustancia_def sustancia_negativa_def by blast

text ‹Sin los axiomas, la negativa es más débil: valdría también para algo que no es
  en nada ni se concibe por nada.›

lemma sustancia_neg_sin_axiomas:
  "sustancia_negativa x ⟹ sustancia x"
  nitpick [show_consts]
  oops


section ‹Todo es sustancia o modo›

text ‹Spinoza lo da por sentado (prop. 4, dem.: «nada hay fuera del entendimiento
  excepto las sustancias y sus afecciones»). Con A1 y A2 solos no sale:›

lemma todo_sustancia_o_modo_sin_premisa:
  "A1 ⟹ A2 ⟹ sustancia x ∨ modo x"
  nitpick [show_consts]
  oops

theorem todo_sustancia_o_modo:
  assumes "A1" "H_ser_en_implica_concebirse" "H_solo_en_si_solo_por_si"
  shows "sustancia x ∨ modo x"
proof (cases "∃y. y ≠ x ∧ En x y")
  case True
  then obtain y where "y ≠ x" "En x y" by blast
  moreover have "SeConcibePor x y"
    using ‹En x y› ‹H_ser_en_implica_concebirse› unfolding H_ser_en_implica_concebirse_def by blast
  ultimately show ?thesis unfolding modo_def by blast
next
  case False
  then have solo: "∀y. En x y ⟶ y = x" by blast
  have "En x x" using ‹A1› False unfolding A1_def by blast
  moreover have "SeConcibePor x x"
    using ‹En x x› ‹H_ser_en_implica_concebirse› unfolding H_ser_en_implica_concebirse_def by blast
  moreover have "∀y. SeConcibePor x y ⟶ y = x"
    using solo ‹H_solo_en_si_solo_por_si› unfolding H_solo_en_si_solo_por_si_def by blast
  ultimately show ?thesis unfolding sustancia_def using solo by blast
qed


section ‹Proposiciones 1-6›

text ‹Prop. 1: La sustancia es anterior_por_naturaleza por naturaleza a sus afecciones.
  Sale solo de las definiciones.›

theorem P1:
  assumes "sustancia s" and "afeccion m s"
  shows "anterior_por_naturaleza s m"
  using assms unfolding sustancia_def afeccion_def anterior_por_naturaleza_def by blast

text ‹Prop. 2: Dos sustancias que tienen atributos distintos no tienen nada en común.
  Spinoza: «es evidente por la def. 3». Con los axiomas solos, no:›

lemma P2_sin_premisa:
  "A1 ⟹ A2 ⟹ A4 ⟹ A5 ⟹ sustancia s ⟹ sustancia t ⟹ s ≠ t ⟹ ¬ AlgoEnComun s t"
  nitpick [show_consts]
  oops

text ‹Con la premisa oculta sí. Y nótese que la hipótesis «atributos distintos»
  sobra: basta con que sean dos sustancias distintas.›

theorem P2:
  assumes "H_conversa_A5" "sustancia s" "sustancia t" "s ≠ t"
  shows "¬ AlgoEnComun s t"
  using assms unfolding H_conversa_A5_def sustancia_def by blast

text ‹Prop. 3: Si dos cosas no tienen nada en común, una no puede ser causa de la otra.
  Esta sí sale limpia de los axiomas 4 y 5, tal como dice Spinoza.›

theorem P3:
  assumes "A4" "A5" "¬ AlgoEnComun x y"
  shows "¬ Causa x y ∧ ¬ Causa y x"
  using assms unfolding A4_def A5_def by blast

text ‹Prop. 5: No puede haber dos o más sustancias con la misma naturaleza o atributo.
  Contramodelo de Leibniz: s con atributos {a, b} y t con {a, c}.›

lemma P5_sin_premisa:
  "A1 ⟹ A2 ⟹ A4 ⟹ A5 ⟹ H_individuacion_por_atributos ⟹ sustancia s ⟹ sustancia t ⟹ TieneAtributo s a ⟹ TieneAtributo t a ⟹ s = t"
  nitpick [show_consts]
  oops

theorem P5:
  assumes "H_individuacion_por_atributos" "H_quien_comparte_uno_comparte_todos" "sustancia s" "sustancia t" "TieneAtributo s a" "TieneAtributo t a"
  shows "s = t"
  using assms unfolding H_individuacion_por_atributos_def H_quien_comparte_uno_comparte_todos_def by blast

text ‹Un camino que Spinoza no toma: con H_conversa_A5 la prop. 5 sale de la prop. 2,
  sin necesitar la premisa que objetó Leibniz.›

theorem P5_por_P2:
  assumes "H_conversa_A5" "H_atributo_compartido_es_comun" "sustancia s" "sustancia t" "TieneAtributo s a" "TieneAtributo t a"
  shows "s = t"
  using assms P2 [of s t] unfolding H_atributo_compartido_es_comun_def by blast

text ‹Prop. 6: Una sustancia no puede ser producida por otra sustancia.
  Prueba de Spinoza: props. 5, 2 y 3.›

theorem P6:
  assumes "H_conversa_A5" "A4" "A5" "sustancia s" "sustancia t" "s ≠ t"
  shows "¬ Causa s t"
  using assms P2 [of s t] P3 [of s t] by blast

text ‹Corolario, prueba alternativa de Spinoza («se demuestra más fácilmente por el
  absurdo»): por el ax. 4 y la def. 3. No necesita ninguna premisa oculta.›

theorem P6_aliter:
  assumes "A4" "sustancia s" "sustancia t" "s ≠ t"
  shows "¬ Causa s t"
  using assms unfolding A4_def sustancia_def by blast

section ‹Proposiciones 7-11: vocabulario y definiciones nuevas›

consts
  EsenciaImplicaExistencia     :: "cosa ⇒ bool"            ― ‹su esencia implica la existencia›
  ConcebibleComoNoExistente :: "cosa ⇒ bool"            ― ‹puede concebirse como no existente›
  Existe                :: "cosa ⇒ bool"            ― ‹existe fuera del concepto›
  Limita                :: "cosa ⇒ cosa ⇒ bool"     ― ‹Limita y x: y limita a x›
  AtributoSeConcibePor            :: "atributo ⇒ atributo ⇒ bool"  ― ‹el atributo a se concibe por b›
  MasRealidad           :: "cosa ⇒ cosa ⇒ bool"     ― ‹x tiene más realidad o ser que y›
  Dios                  :: cosa
  Voluntad              :: cosa

text ‹Def. 1: «Por causa de sí entiendo aquello cuya esencia implica la existencia».›

definition D1 :: bool where
  "D1 ≡ ∀x. Causa x x ⟷ EsenciaImplicaExistencia x"

text ‹Def. 2: «Se dice finita en su género aquella cosa que puede ser limitada por otra
  de la misma naturaleza». «Misma naturaleza» se lee como compartir un atributo.›

definition finita :: "cosa ⇒ bool" where
  "finita x ≡ ∃y. y ≠ x ∧ Limita y x ∧ (∃a. TieneAtributo y a ∧ TieneAtributo x a)"

text ‹Def. 4: el atributo es lo que el entendimiento percibe de una sustancia como
  constitutivo de su esencia. Aquí: solo las sustancias tienen atributos.›

definition D4 :: bool where
  "D4 ≡ ∀x a. TieneAtributo x a ⟶ sustancia x"

text ‹Def. 6: Dios es un ente absolutamente infinito, una sustancia que consta de infinitos
  atributos. En un modelo finito, «infinitos» se lee como «todos».›

definition D6 :: bool where
  "D6 ≡ sustancia Dios ∧ (∀a. TieneAtributo Dios a)"

text ‹Ax. 7: «Si una cosa puede concebirse como no existente, su esencia no implica
  la existencia».›

definition A7 :: bool where
  "A7 ≡ ∀x. ConcebibleComoNoExistente x ⟶ ¬ EsenciaImplicaExistencia x"

text ‹H_todo_tiene_causa: todo tiene una causa, sea otra cosa o ella misma. Spinoza la usa en la
  prop. 7 sin enunciarla; la formula después, en la 2.ª prueba de la prop. 11.›

definition H_todo_tiene_causa :: bool where
  "H_todo_tiene_causa ≡ ∀x. ∃y. Causa y x"


section ‹Proposiciones 7-11›

text ‹Prop. 6, corolario: una sustancia no puede ser producida por nada distinto de
  ella (no solo por otra sustancia). Con la def. 3 endurecida basta el ax. 4.›

theorem P6_cor:
  assumes "A4" "sustancia s" "y ≠ s"
  shows "¬ Causa y s"
  using assms unfolding A4_def sustancia_def by blast

text ‹Prop. 7: A la naturaleza de la sustancia pertenece existir.
  Spinoza: no la produce otra cosa (cor. prop. 6), «luego será causa de sí». Ese «luego»
  supone que todo tiene causa. Sin H_todo_tiene_causa, una sustancia sin causa alguna:›

lemma P7_sin_premisa:
  "A4 ⟹ D1 ⟹ sustancia s ⟹ EsenciaImplicaExistencia s"
  nitpick [show_consts]
  oops

theorem P7:
  assumes "A4" "D1" "H_todo_tiene_causa" "sustancia s"
  shows "EsenciaImplicaExistencia s"
proof -
  obtain y where c: "Causa y s" using ‹H_todo_tiene_causa› unfolding H_todo_tiene_causa_def by blast
  then have "y = s" using P6_cor [of s y] ‹A4› ‹sustancia s› by blast
  then show ?thesis using c ‹D1› unfolding D1_def by blast
qed

text ‹Prop. 8: Toda sustancia es necesariamente infinita.
  Spinoza: si fuera finita, la limitaría otra de la misma naturaleza, que también
  existiría (prop. 7), y habría dos sustancias del mismo atributo, contra la prop. 5.
  Aquí la prop. 7 no hace falta: en esta lógica, cuantificar sobre «otra cosa» ya
  supone que existe. Lo que sí hace falta es todo lo que necesitaba la prop. 5.›

theorem P8:
  assumes "D4" "H_individuacion_por_atributos" "H_quien_comparte_uno_comparte_todos" "sustancia s"
  shows "¬ finita s"
proof
  assume "finita s"
  then obtain y a where "y ≠ s" "TieneAtributo y a" "TieneAtributo s a" unfolding finita_def by blast
  moreover have "sustancia y" using ‹D4› ‹TieneAtributo y a› unfolding D4_def by blast
  ultimately show False using P5 [of y s a] assms by blast
qed

text ‹Prop. 9: Cuanta más realidad o ser tiene una cosa, tantos más atributos le
  competen. Spinoza: «es evidente por la def. 4». Pero la def. 4 no habla de grados
  de realidad:›

lemma P9_sin_premisa:
  "D4 ⟹ MasRealidad x y ⟹ {a. TieneAtributo y a} ⊂ {a. TieneAtributo x a}"
  nitpick [show_consts]
  oops

text ‹Para que salga hay que identificar grado de realidad con cantidad de atributos,
  y entonces la prop. 9 es esa misma premisa.›

definition H_realidad_es_atributos :: bool where
  "H_realidad_es_atributos ≡ ∀x y. MasRealidad x y ⟶ {a. TieneAtributo y a} ⊂ {a. TieneAtributo x a}"

theorem P9:
  assumes "H_realidad_es_atributos" "MasRealidad x y"
  shows "{a. TieneAtributo y a} ⊂ {a. TieneAtributo x a}"
  using assms unfolding H_realidad_es_atributos_def by blast

text ‹Prop. 10: Cada atributo de una sustancia debe concebirse por sí.
  Spinoza: por la def. 4 y la def. 3. Hace falta un puente entre atributos y
  sustancias: si un atributo se concibe por otro, su sustancia se concibe por la
  sustancia del otro.›

definition H_puente_atributo_sustancia :: bool where
  "H_puente_atributo_sustancia ≡ ∀s t a b. TieneAtributo s a ∧ TieneAtributo t b ∧ AtributoSeConcibePor a b ⟶ SeConcibePor s t"

text ‹Lo que sí sale: un atributo solo puede concebirse por atributos de su misma
  sustancia.›

theorem P10_debil:
  assumes "H_puente_atributo_sustancia" "D4" "TieneAtributo s a" "TieneAtributo t b" "AtributoSeConcibePor a b"
  shows "s = t"
  using assms unfolding H_puente_atributo_sustancia_def D4_def sustancia_def by blast

text ‹Lo que no sale: que un atributo no se conciba por otro de la misma sustancia
  (el escolio: atributos «realmente distintos»). Dios con dos atributos, uno
  concebido por el otro:›

lemma P10_sin_premisa:
  "H_puente_atributo_sustancia ⟹ D4 ⟹ D6 ⟹ TieneAtributo s a ⟹ AtributoSeConcibePor a b ⟹ a = b"
  nitpick [show_consts]
  oops

text ‹Prop. 11: Dios, o sea, una sustancia que consta de infinitos atributos, existe
  necesariamente. 1.ª prueba: si se niega, concíbase a Dios como no existente; por el
  ax. 7 su esencia no implicaría la existencia, contra la prop. 7.›

theorem P11:
  assumes "A4" "D1" "H_todo_tiene_causa" "A7" "D6"
  shows "¬ ConcebibleComoNoExistente Dios"
proof
  assume "ConcebibleComoNoExistente Dios"
  then have "¬ EsenciaImplicaExistencia Dios" using ‹A7› unfolding A7_def by blast
  moreover have "EsenciaImplicaExistencia Dios"
    using P7 [of Dios] assms unfolding D6_def by blast
  ultimately show False by blast
qed

text ‹La prueba da que Dios no puede concebirse como no existente. Que exista fuera
  del concepto no se sigue de nada de lo anterior_por_naturaleza:›

lemma P11_existencia_real:
  "A4 ⟹ D1 ⟹ H_todo_tiene_causa ⟹ A7 ⟹ D6 ⟹ Existe Dios"
  nitpick [show_consts]
  oops

text ‹Además, en HOL la constante Dios denota algo del dominio desde que se declara:
  la existencia del sujeto ya está puesta en el lenguaje antes de la prueba. Es el
  viejo reproche al argumento ontológico. Spinoza cerraría el paso con el ax. 6 (la
  idea verdadera debe convenir con su objeto), y ese es el punto que discutió Kant.›

section ‹Proposiciones 12-15›

consts
  EsParteDe :: "cosa ⇒ cosa ⇒ bool"   ― ‹EsParteDe p x: p es parte de x›

definition divisible :: "cosa ⇒ bool" where
  "divisible x ≡ ∃p. p ≠ x ∧ EsParteDe p x"

text ‹Spinoza, en el primer cuerno del dilema de la prop. 12: el todo «podría ser y
  concebirse sin sus partes, lo cual nadie dudará que es absurdo». No es una premisa
  oculta: la enuncia, pero como evidente.›

definition H_todo_por_sus_partes :: bool where
  "H_todo_por_sus_partes ≡ ∀p x. EsParteDe p x ⟶ SeConcibePor x p"

text ‹Prop. 12: No puede concebirse con verdad ningún atributo de la sustancia del que
  se siga que la sustancia puede dividirse. Spinoza la prueba con un dilema (las partes
  conservan o no la naturaleza de sustancia) que recorre las props. 2, 5, 6, 7, 8 y 10.
  Aquí basta la premisa anterior_por_naturaleza más la def. 3: lo que se concibe por sus partes no se
  concibe solo por sí.›

lemma P12_sin_premisa:
  "A1 ⟹ A2 ⟹ A4 ⟹ A5 ⟹ sustancia s ⟹ ¬ divisible s"
  nitpick [show_consts]
  oops

theorem P12:
  assumes "H_todo_por_sus_partes" "sustancia s"
  shows "¬ divisible s"
  using assms unfolding H_todo_por_sus_partes_def sustancia_def divisible_def by blast

text ‹Prop. 13: La sustancia absolutamente infinita es indivisible.›

theorem P13:
  assumes "H_todo_por_sus_partes" "D6"
  shows "¬ divisible Dios"
  using assms P12 [of Dios] unfolding D6_def by blast

text ‹Prop. 14: Fuera de Dios no puede darse ni concebirse ninguna sustancia.
  Spinoza: Dios tiene todos los atributos (def. 6); otra sustancia tendría que
  explicarse por alguno de ellos y habría dos sustancias del mismo atributo, contra la
  prop. 5. Ese «tendría que explicarse por alguno» supone que toda sustancia tiene al
  menos un atributo. Sin esa premisa, una sustancia sin atributos convive con Dios:›

definition H_sustancia_con_atributo :: bool where
  "H_sustancia_con_atributo ≡ ∀s. sustancia s ⟶ (∃a. TieneAtributo s a)"

lemma P14_sin_premisa:
  "D4 ⟹ D6 ⟹ H_individuacion_por_atributos ⟹ H_quien_comparte_uno_comparte_todos ⟹ sustancia s ⟹ s = Dios"
  nitpick [show_consts]
  oops

theorem P14:
  assumes "D6" "H_sustancia_con_atributo" "H_individuacion_por_atributos" "H_quien_comparte_uno_comparte_todos" "sustancia s"
  shows "s = Dios"
proof -
  obtain a where "TieneAtributo s a" using ‹H_sustancia_con_atributo› ‹sustancia s› unfolding H_sustancia_con_atributo_def by blast
  moreover have "sustancia Dios" "TieneAtributo Dios a" using ‹D6› unfolding D6_def by blast+
  ultimately show ?thesis using P5 [of s Dios a] assms by blast
qed

text ‹Corolario 1: Dios es único. No necesita H_sustancia_con_atributo: en HOL todo tipo tiene
  al menos un elemento, así que dos sustancias con todos los atributos comparten uno.
  (Ese «al menos un atributo» lo pone el lenguaje, no Spinoza.)›

theorem P14_cor1:
  assumes "D6" "H_individuacion_por_atributos" "H_quien_comparte_uno_comparte_todos" "sustancia x" "∀a. TieneAtributo x a"
  shows "x = Dios"
proof -

  moreover have "TieneAtributo x undefined" using assms by blast
  ultimately show ?thesis using P5 [of x Dios undefined] assms by blast
qed

text ‹Prop. 15: Todo lo que es, es en Dios, y sin Dios nada puede ser ni concebirse.

definition H_modo_en_sustancia :: bool where

lemma P15_sin_premisa:
  "A1 ⟹ H_ser_en_implica_concebirse ⟹ H_solo_en_si_solo_por_si ⟹ D6 ⟹ H_sustancia_con_atributo ⟹ H_individuacion_por_atributos ⟹ H_quien_comparte_uno_comparte_todos ⟹ En x Dios"

  oops

theorem P15:
  assumes "A1" "H_ser_en_implica_concebirse" "H_solo_en_si_solo_por_si" "H_modo_en_sustancia" "D6" "H_sustancia_con_atributo" "H_individuacion_por_atributos" "H_quien_comparte_uno_comparte_todos"

    then show "En x Dios" using ‹sustancia x› unfolding sustancia_def by blast

    then obtain y where "y ≠ x" "En x y" unfolding modo_def by blast

    then have "y = Dios" using P14 assms by blast
    then show "En x Dios" using ‹En x y› by blast

  then show ?thesis using ‹H_ser_en_implica_concebirse› unfolding H_ser_en_implica_concebirse_def by blast
qed


section ‹Las partes de los modos›

text ‹El primer modelo de compatibilidad con modos divisibles hacía de Dios una parte
  de un modo. Spinoza no lo admitiría: las partes de un modo son modos.›

definition H_partes_de_modos :: bool where
  "H_partes_de_modos ≡ ∀p x. modo x ∧ EsParteDe p x ∧ p ≠ x ⟶ modo p"

text ‹Ninguna cosa es a la vez sustancia y modo.›

lemma sustancia_no_modo:
  assumes "sustancia x"
  shows "¬ modo x"
  using assms unfolding sustancia_def modo_def by blast

text ‹Pero con esta premisa, junto a «ser en = concebirse por» y «un modo solo es en
  sustancias», ningún modo puede tener partes. Una parte p del modo m: m se concibe
  por p (el todo por sus partes), luego es en p, luego p es sustancia; y a la vez p es
  modo. Los cuerpos quedarían sin partes, contra lo que Spinoza dice de ellos en la
  Parte II.›

theorem modos_indivisibles:
  assumes "H_todo_por_sus_partes" "H_ser_en_igual_concebirse" "H_modo_en_sustancia" "H_partes_de_modos" "modo m"
  shows "¬ divisible m"
proof
  assume "divisible m"
  then obtain p where "p ≠ m" "EsParteDe p m" unfolding divisible_def by blast
  then have "En m p"
    using ‹H_todo_por_sus_partes› ‹H_ser_en_igual_concebirse› unfolding H_todo_por_sus_partes_def H_ser_en_igual_concebirse_def by blast
  then have "sustancia p"
    using ‹H_modo_en_sustancia› ‹modo m› ‹p ≠ m› unfolding H_modo_en_sustancia_def by blast
  moreover have "modo p"
    using ‹H_partes_de_modos› ‹modo m› ‹EsParteDe p m› ‹p ≠ m› unfolding H_partes_de_modos_def by blast
  ultimately show False using sustancia_no_modo by blast
qed

text ‹El mismo mecanismo choca con la causalidad entre modos: por el ax. 4 el efecto se
  concibe por su causa, luego es en ella, luego la causa es sustancia, luego es Dios.
  Ningún modo podría causar otro modo, contra la prop. 28 (cada cosa finita es causada
  por otra cosa finita).›

theorem modos_causados_solo_por_Dios:
  assumes "A4" "H_ser_en_igual_concebirse" "H_modo_en_sustancia" "D6" "H_sustancia_con_atributo" "H_individuacion_por_atributos" "H_quien_comparte_uno_comparte_todos"
      "modo m" "Causa y m" "y ≠ m"
  shows "y = Dios"
proof -
  have "En m y"
    using ‹A4› ‹H_ser_en_igual_concebirse› ‹Causa y m› unfolding A4_def H_ser_en_igual_concebirse_def by blast
  then have "sustancia y"
    using ‹H_modo_en_sustancia› ‹modo m› ‹y ≠ m› unfolding H_modo_en_sustancia_def by blast
  then show ?thesis using P14 assms by blast
qed

text ‹El culpable es H_ser_en_igual_concebirse: para las sustancias, «ser en» y «concebirse por»
  coinciden; para los modos, no. Un modo es en Dios, pero se concibe también por sus
  causas y por sus partes, que son modos.›

text ‹Por eso, desde aquí la teoría usa la versión debilitada (H_ser_en_implica_concebirse y
  H_solo_en_si_solo_por_si). Los dos teoremas anteriores siguen siendo válidos, pero solo
  para la premisa fuerte, que ya no se usa.›


section ‹«En sí o en otro», «por sí o por otro»: ¿excluyentes?›

text ‹El ax. 1 dice que todo es en sí o en otra cosa, pero no que no pueda ser las dos.
  En el modelo con modos divisibles, todo era «en sí» además de ser en Dios.
  Lectura excluyente:›

definition A1_excluyente :: bool where
  "A1_excluyente ≡ ∀x. (En x x ∧ ¬ (∃y. y ≠ x ∧ En x y)) ∨ (¬ En x x ∧ (∃y. y ≠ x ∧ En x y))"

lemma A1_excl_implica_A1: "A1_excluyente ⟹ A1"
  unfolding A1_excluyente_def A1_def by blast

text ‹Lo mismo con la concepción: lo que se concibe por otra cosa (def. 5) no se
  concibe por sí (def. 3). Spinoza contrapone «per se» y «per aliud» como si se
  excluyeran, pero no lo dice.›

definition H_por_si_excluye_por_otro :: bool where
  "H_por_si_excluye_por_otro ≡ ∀x. (∃y. y ≠ x ∧ SeConcibePor x y) ⟶ ¬ SeConcibePor x x"

text ‹Prop. 24: La esencia de las cosas producidas por Dios no implica la existencia.
  Spinoza: «es evidente por la def. 1», porque lo que la implica es causa de sí.
  Pero en el modelo anterior_por_naturaleza había un modo causa de sí. Sin premisas, sale de nuevo:›

lemma P24_sin_premisa:
  "A4 ⟹ D1 ⟹ modo m ⟹ ¬ EsenciaImplicaExistencia m"
  nitpick [show_consts]
  oops

text ‹La lectura excluyente del ax. 1 no basta: un modo que no es en sí, pero que se
  concibe a la vez por sí y por Dios, puede seguir siendo causa de sí.›

lemma P24_solo_A1_excl:
  "A1_excluyente ⟹ A4 ⟹ D1 ⟹ H_ser_en_implica_concebirse ⟹ modo m ⟹ ¬ EsenciaImplicaExistencia m"
  nitpick [show_consts]
  oops

text ‹Lo que hace falta es la exclusividad de la concepción: si su esencia implicara
  la existencia, sería causa de sí (def. 1), luego se concebiría por sí (ax. 4); pero

theorem P24:
  assumes "A4" "D1" "H_por_si_excluye_por_otro" "modo m"

qed


section ‹Proposiciones 16-36: Dios como causa (primera tanda)›

text ‹Def. 7: «Se dice libre la cosa que existe por la sola necesidad de su naturaleza
  y es determinada a obrar por sí sola». Leemos «ser determinado por» como «ser causado
  por».›

definition libre :: "cosa ⇒ bool" where
  "libre x ≡ EsenciaImplicaExistencia x ∧ (∀y. Causa y x ⟶ y = x)"

text ‹Def. 8: «Por eternidad entiendo la existencia misma, en cuanto se concibe que se
  sigue necesariamente de la sola definición de la cosa eterna».›

definition eterno :: "cosa ⇒ bool" where
  "eterno x ≡ EsenciaImplicaExistencia x"

text ‹Un atributo es eterno si lo es la sustancia cuya esencia expresa.›

definition atributo_eterno :: "atributo ⇒ bool" where
  "atributo_eterno a ≡ ∀s. TieneAtributo s a ⟶ eterno s"

text ‹Causa inmanente: el efecto es en la causa. Causa transitiva: el efecto no es en ella.›

definition causa_inmanente :: "cosa ⇒ cosa ⇒ bool" where
  "causa_inmanente y x ≡ Causa y x ∧ En x y"

definition causa_transitiva :: "cosa ⇒ cosa ⇒ bool" where
  "causa_transitiva y x ≡ Causa y x ∧ ¬ En x y"

text ‹Prop. 33, esc. 1: una cosa se dice necesaria por su esencia o por su causa;
  contingente, si no lo es por ninguna de las dos.›

definition contingente :: "cosa ⇒ bool" where
  "contingente x ≡ ¬ EsenciaImplicaExistencia x ∧ ¬ (∃y. Causa y x)"

text ‹Las premisas de la prop. 15, agrupadas para no repetirlas.›

text ‹H_inherencia_es_causalidad: si una cosa es en otra, esa otra es su causa.

section ‹Modos infinitos y doble eternidad (props. 21-23 y 28)›

consts
  ModoDeAtributo          :: "cosa ⇒ atributo ⇒ bool"  ― ‹m es un modo del atributo a›
  SigueDeAtributoAbsoluto :: "cosa ⇒ atributo ⇒ bool"  ― ‹x se sigue de la naturaleza absoluta de a›
  SigueDeModificacion     :: "cosa ⇒ cosa ⇒ bool"      ― ‹x se sigue de un atributo modificado por y›

text ‹Def. 2 para modos: es finito el modo al que limita otro modo del mismo atributo.›

definition finita_modo :: "cosa ⇒ bool" where
  "finita_modo x ≡ ∃y. y ≠ x ∧ Limita y x ∧ (∃a. ModoDeAtributo y a ∧ ModoDeAtributo x a)"

text ‹Premisas que dan sentido al vocabulario nuevo.
  H_modo_de_atributo_es_modo: lo que es modo de un atributo es un modo.
  H_todo_modo_de_algun_atributo: y todo modo lo es de algún atributo (cf. II, prop. 6).
  H_limitar_es_determinar: lo que limita a una cosa la determina, es decir, es causa
    suya (def. 2 leída con la def. 7: lo limitado está «coaccionado» por otro).
  H_lo_que_limita_es_finito: lo que limita a otra cosa es a su vez finito (como el
    cuerpo mayor del ejemplo de la def. 2, que otro mayor limita).
  H_absoluto_solo_de_Dios: lo que se sigue de la naturaleza absoluta de un atributo es
    modo de ese atributo y no tiene más causa que Dios (se sigue sin intermediarios).
  H_mediato_de_Dios_y_su_modificacion: lo que se sigue de un atributo modificado por y
    tiene por causa a y, que es un modo, y no tiene más causas que Dios e y. (Sin «que es
    un modo», Nitpick hacía seguirse un modo de un atributo «modificado por Dios».)
  H_dos_maneras_de_seguirse: todo modo se sigue de Dios de una de las dos maneras.
  H_de_lo_finito_lo_finito: lo que se sigue de una modificación finita es finito.›

definition H_modo_de_atributo_es_modo :: bool where
  "H_modo_de_atributo_es_modo ≡ ∀y a. ModoDeAtributo y a ⟶ modo y"

definition H_todo_modo_de_algun_atributo :: bool where
  "H_todo_modo_de_algun_atributo ≡ ∀m. modo m ⟶ (∃a. ModoDeAtributo m a)"

definition H_limitar_es_determinar :: bool where
  "H_limitar_es_determinar ≡ ∀x y. Limita y x ⟶ Causa y x"

definition H_lo_que_limita_es_finito :: bool where
  "H_lo_que_limita_es_finito ≡ ∀x y. Limita y x ∧ y ≠ x ⟶ finita_modo y"

definition H_absoluto_solo_de_Dios :: bool where
  "H_absoluto_solo_de_Dios ≡ ∀x a. SigueDeAtributoAbsoluto x a
     ⟶ ModoDeAtributo x a ∧ (∀z. Causa z x ⟶ z = Dios)"

definition H_mediato_de_Dios_y_su_modificacion :: bool where
  "H_mediato_de_Dios_y_su_modificacion ≡ ∀x y. SigueDeModificacion x y
     ⟶ modo y ∧ Causa y x ∧ (∀z. Causa z x ⟶ z = Dios ∨ z = y)"

definition H_dos_maneras_de_seguirse :: bool where
  "H_dos_maneras_de_seguirse ≡ ∀m. modo m
     ⟶ (∃a. SigueDeAtributoAbsoluto m a) ∨ (∃y. SigueDeModificacion m y)"

definition H_de_lo_finito_lo_finito :: bool where
  "H_de_lo_finito_lo_finito ≡ ∀m y. SigueDeModificacion m y ∧ finita_modo y ⟶ finita_modo m"

text ‹Doble eternidad. La def. 8, como la leímos, es la eternidad por la esencia
  (eterno: la esencia implica la existencia). Spinoza dice además que los modos
  infinitos son eternos «por el atributo» (prop. 21): porque se siguen, inmediata o
  mediatamente, de un atributo eterno.›

definition eterno_por_atributo :: "cosa ⇒ bool" where
  "eterno_por_atributo x ≡ (∃a. SigueDeAtributoAbsoluto x a)
     ∨ (∃y a. SigueDeModificacion x y ∧ SigueDeAtributoAbsoluto y a)"

lemma ningun_modo_eterno_por_esencia:
  assumes "A4" "D1" "H_por_si_excluye_por_otro" "modo x"

section ‹Mente y cuerpo (II, props. 6 y 7; III, prop. 2)›

consts
  Pensamiento :: atributo
  Extension   :: atributo

text ‹Premisas.
  H_atributos_distintos: pensamiento y extensión son atributos distintos.
  H_modo_se_concibe_por_su_atributo: un modo de un atributo solo se concibe, fuera de
    Dios, por cosas de ese mismo atributo. Es la versión para los modos de la prop. 10
    fuerte, y la razón que da Spinoza en la demostración de II, 6: los modos de cada
    atributo implican el concepto de su atributo y no el de otro.
  H_un_atributo_por_modo: cada modo lo es de un solo atributo. Choca con la lectura
    literal del escolio de II, 7 («un modo de la extensión y la idea de ese modo son una
    y la misma cosa»): aquí el cuerpo y su idea son dos cosas del dominio.
  H_paralelismo: si una cosa causa otra, la idea de la primera causa la idea de la
    segunda (II, 7: el orden y la conexión de las ideas es el de las cosas).
  H_ideas_del_pensamiento: las ideas son modos del pensamiento.›

definition H_atributos_distintos :: bool where
  "H_atributos_distintos ≡ Pensamiento ≠ Extension"

definition H_modo_se_concibe_por_su_atributo :: bool where
  "H_modo_se_concibe_por_su_atributo ≡ ∀x y b. ModoDeAtributo y b ∧ SeConcibePor y x ∧ x ≠ Dios
     ⟶ ModoDeAtributo x b"

definition H_un_atributo_por_modo :: bool where
  "H_un_atributo_por_modo ≡ ∀x a b. ModoDeAtributo x a ∧ ModoDeAtributo x b ⟶ a = b"

definition H_paralelismo :: bool where
  "H_paralelismo ≡ ∀x y i j. Causa x y ∧ IdeaDe i x ∧ IdeaDe j y ⟶ Causa i j"

definition H_ideas_del_pensamiento :: bool where

section ‹El «en cuanto» (quatenus): cuerpo e idea, una misma cosa›

text ‹Spinoza dice que un modo de la extensión y su idea son «una y la misma cosa,
  expresada de dos maneras» (II, 7, esc.), y aun así que el cuerpo no causa la mente
  (III, 2). Lo hace compatible con el «en cuanto»: una cosa causa algo en cuanto se la
  considera bajo un atributo. Relativizamos la causalidad y la concepción a los
  atributos, y dejamos que una cosa sea modo de varios (sin H_un_atributo_por_modo).›

consts
  CausaBajo     :: "atributo ⇒ atributo ⇒ cosa ⇒ cosa ⇒ bool"
    ― ‹CausaBajo a b x y: x, en cuanto se le considera bajo a, causa y en cuanto bajo b›
  SeConcibeBajo :: "atributo ⇒ atributo ⇒ cosa ⇒ cosa ⇒ bool"
    ― ‹SeConcibeBajo b a y x: y, en cuanto bajo b, se concibe por x en cuanto bajo a›

text ‹Premisas, versiones «en cuanto» de las anteriores.
  A4_quatenus: el efecto, en cuanto b, se concibe por la causa en cuanto a (ax. 4).
  H_atributo_por_si_quatenus: lo que se concibe bajo un atributo solo se concibe por
    algo bajo ese mismo atributo (prop. 10 fuerte, «sin el otro»).
  H_en_cuanto_requiere_atributo: para causar o ser causado en cuanto a, hay que tener a
    (Dios, como causa, los tiene todos).
  H_mismo_orden: si x causa y bajo un atributo, lo causa bajo cualquier otro que ambos
    tengan (II, 7, esc.: «un mismo orden, una misma conexión de causas»).
  H_causa_relativa_es_causa: causar en cuanto es causar.›

definition A4_quatenus :: bool where
  "A4_quatenus ≡ ∀a b x y. CausaBajo a b x y ⟶ SeConcibeBajo b a y x"

definition H_atributo_por_si_quatenus :: bool where
  "H_atributo_por_si_quatenus ≡ ∀a b x y. SeConcibeBajo b a y x ⟶ a = b"

definition H_en_cuanto_requiere_atributo :: bool where
  "H_en_cuanto_requiere_atributo ≡ ∀a b x y. CausaBajo a b x y
     ⟶ (x = Dios ∨ ModoDeAtributo x a) ∧ ModoDeAtributo y b"

definition H_mismo_orden :: bool where
  "H_mismo_orden ≡ ∀a b x y. CausaBajo a a x y ∧ ModoDeAtributo x b ∧ ModoDeAtributo y b
     ⟶ CausaBajo b b x y"

definition H_causa_relativa_es_causa :: bool where
  "H_causa_relativa_es_causa ≡ ∀a b x y. CausaBajo a b x y ⟶ Causa x y"

  oops
